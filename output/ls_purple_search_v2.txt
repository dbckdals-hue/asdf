단기 = (highest(h, 9) + lowest(l, 9)) / 2;
중기 = (highest(h, 26) + lowest(l, 26)) / 2;
SA = (단기 + 중기) / 2;
SB = (highest(h, 52) + lowest(l, 52)) / 2;

// 보라색 배지 기준 검색식 (상승 삼각형) - 스크립트 v5.28 보라 기준과 동일
// 사용 함수: highest, lowest, max, abs, 사칙연산, 비교, &&, || 만 사용
// [삼각형] 폭1은 SA가 SB 위로 1봉만 올라갔다 내려옴, 폭2는 2봉
// [왼쪽평탄구간] 삼각형 왼쪽 SB 불변 2봉 이상 (왼쪽 기준봉 포함)
// [오른쪽평탄구간] 삼각형 오른쪽 기준봉(현재봉)의 SB 동일 1봉 (Flat 식에 포함)
// [왼쪽구름층] 삼각형 왼쪽 8봉 중 반대색 구름(SB가 SA보다 위)이 4봉 이상
// [등록시점] 삼각형이 완성되는 현재봉(SA가 SB 아래로 복귀한 봉)에서 즉시 신호
// ================================================================

// [왼쪽평탄구간]+[오른쪽평탄구간] SB 불변 봉수
// 폭1 삼각형은 현재봉, 삼각형 1봉, 왼쪽 기준봉이 SB 동일(3봉)이고 왼쪽으로 1봉 더(4봉) = FlatB2
// 폭2 삼각형은 현재봉, 삼각형 2봉, 왼쪽 기준봉이 SB 동일(4봉)이고 왼쪽으로 1봉 더(5봉) = FlatB3
// 왼쪽 평탄 최소 봉수를 늘리려면 같은 방식으로 FlatB4 등을 추가해서 Purple 식에 연결
FlatB1 = (SB(2) == SB(1)) && (SB(1) == SB);
FlatB2 = (SB(3) == SB(2)) && FlatB1;
FlatB3 = (SB(4) == SB(3)) && FlatB2;

// [삼각형] SA가 SB 위로 올라갔다 내려온 모양
// 폭1은 SA(2)가 왼쪽 기준봉, SA(1)이 삼각형, SA가 오른쪽 기준봉(현재봉)
// 폭2는 SA(3)이 왼쪽 기준봉, SA(2)와 SA(1)이 삼각형 2봉, SA가 오른쪽 기준봉
// [등록시점] 현재봉이 SA <= SB 로 돌아오는 순간 true 이므로 삼각형 완성 즉시 신호
Tri1 = (SA(2) <= SB(2)) && (SA(1) > SB(1)) && (SA <= SB);
Tri2 = (SA(3) <= SB(3)) && (SA(2) > SB(2)) && (SA(1) > SB(1)) && (SA <= SB);

// [삼각형] 꼭짓점이 뾰족한지 (폭2에만 적용, 폭1은 항상 통과)
// 두 삼각형 봉의 높이 차이가 큰 쪽 높이의 이 비율 이상이어야 뾰족
// 올리면(예 0.25) 더 뾰족한 것만, 내리면(예 0.05) 납작해도 통과
SharpRatio = 0.15;
Hgt1 = SA(1) - SB(1);
Hgt2 = SA(2) - SB(2);
HgtMax = max(max(Hgt1, Hgt2), 0.0001);
Sharp2 = (abs(Hgt1 - Hgt2) / HgtMax) >= SharpRatio;

// [왼쪽구름층] 봉마다 반대색 구름(SB가 SA보다 위)이면 1, 아니면 0
// (SB-SA) 를 그 절대값(최소 0.0001)으로 나누면 양수 1, 음수 -1, 0이면 0 이 되고
// max(값, 0) 으로 구름 있는 봉만 1이 된다
Cl2 = max((SB(2) - SA(2)) / max(abs(SB(2) - SA(2)), 0.0001), 0);
Cl3 = max((SB(3) - SA(3)) / max(abs(SB(3) - SA(3)), 0.0001), 0);
Cl4 = max((SB(4) - SA(4)) / max(abs(SB(4) - SA(4)), 0.0001), 0);
Cl5 = max((SB(5) - SA(5)) / max(abs(SB(5) - SA(5)), 0.0001), 0);
Cl6 = max((SB(6) - SA(6)) / max(abs(SB(6) - SA(6)), 0.0001), 0);
Cl7 = max((SB(7) - SA(7)) / max(abs(SB(7) - SA(7)), 0.0001), 0);
Cl8 = max((SB(8) - SA(8)) / max(abs(SB(8) - SA(8)), 0.0001), 0);
Cl9 = max((SB(9) - SA(9)) / max(abs(SB(9) - SA(9)), 0.0001), 0);
Cl10 = max((SB(10) - SA(10)) / max(abs(SB(10) - SA(10)), 0.0001), 0);
// 폭1은 왼쪽 8봉이 2~9봉 전, 폭2는 3~10봉 전
LeftCl1 = Cl2 + Cl3 + Cl4 + Cl5 + Cl6 + Cl7 + Cl8 + Cl9;
LeftCl2 = Cl3 + Cl4 + Cl5 + Cl6 + Cl7 + Cl8 + Cl9 + Cl10;
// 8봉 중 몇 봉 이상 구름이 있어야 하는지, 올리면(예 6) 엄격, 내리면(예 3) 느슨
LeftClMin = 4;

// 조합
// 폭1은 왼쪽 평탄 2봉(FlatB2) + 삼각형 + 왼쪽구름 8봉 중 4봉 이상
// 폭2는 왼쪽 평탄 2봉(FlatB3) + 삼각형 + 뾰족 + 왼쪽구름 8봉 중 4봉 이상
// 기존 검색식의 폭3 삼각형(Bull_3)은 스크립트에 없는 형태라 제외
Purple1 = FlatB2 && Tri1 && (LeftCl1 >= LeftClMin);
Purple2 = FlatB3 && Tri2 && Sharp2 && (LeftCl2 >= LeftClMin);
PurpleTri = Purple1 || Purple2;
