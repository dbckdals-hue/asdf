단기 = (highest(h, 9) + lowest(l, 9)) / 2;
중기 = (highest(h, 26) + lowest(l, 26)) / 2;
SA = (단기 + 중기) / 2;
SB = (highest(h, 52) + lowest(l, 52)) / 2;

// ================================================================
// 보라색 배지 기준 검색식 (상승 삼각형) - 스크립트 v5.28 보라 기준과 동일
//  [삼각형]        폭1: SA가 SB 위로 1봉만 올라갔다 내려옴 / 폭2: 2봉 올라갔다 내려옴
//  [왼쪽평탄구간]  삼각형 왼쪽 SB 불변 2봉 이상 (왼쪽 기준봉 포함)
//  [오른쪽평탄구간]삼각형 오른쪽 기준봉(현재봉)의 SB 동일 1봉 (Flat 식에 포함)
//  [왼쪽구름층]    삼각형 왼쪽 8봉 중 반대색 구름(SB>SA)이 4봉 이상
//  [등록시점]      삼각형이 완성되는 현재봉(SA<=SB 로 복귀한 봉)에서 즉시 신호
// ================================================================

// ---- [왼쪽평탄구간]+[오른쪽평탄구간] SB 불변 봉수 -------------------
// 폭1 삼각형: 현재봉(오른쪽 기준봉), 삼각형 1봉, 왼쪽 기준봉 = 3봉 SB 동일(기본)
//             + 왼쪽으로 1봉 더 = 4봉 동일(보라 기준 왼쪽 평탄 2봉) -> FlatB2
// 폭2 삼각형: 현재봉, 삼각형 2봉, 왼쪽 기준봉 = 4봉 동일(기본)
//             + 왼쪽으로 1봉 더 = 5봉 동일(왼쪽 평탄 2봉) -> FlatB3
// 왼쪽 평탄 최소 봉수를 늘리려면 FlatB4, FlatB5... 를 같은 방식으로 추가해서 Bull 식에 연결.
FlatB1 = (SB(2) == SB(1)) && (SB(1) == SB);
FlatB2 = (SB(3) == SB(2)) && FlatB1;
FlatB3 = (SB(4) == SB(3)) && FlatB2;

// ---- [삼각형] SA가 SB 위로 올라갔다 내려온 모양 ----------------------
// 폭1: SA(2)<=SB(2) 왼쪽 기준봉 / SA(1)>SB(1) 삼각형 / SA<=SB 오른쪽 기준봉(현재봉)
// 폭2: SA(3)<=SB(3) 왼쪽 기준봉 / SA(2),SA(1)>SB 삼각형 2봉 / SA<=SB 오른쪽 기준봉
// ([등록시점] 현재봉이 SA<=SB 로 돌아오는 순간 true -> 삼각형 완성 즉시 신호)
삼각1 = (SA(2) <= SB(2)) && (SA(1) > SB(1)) && (SA <= SB);
삼각2 = (SA(3) <= SB(3)) && (SA(2) > SB(2)) && (SA(1) > SB(1)) && (SA <= SB);

// ---- [삼각형] 꼭짓점이 뾰족한지 (폭2 에만 적용, 폭1 은 항상 통과) --------
// 두 삼각형 봉의 높이 차이가 큰 쪽 높이의 이 비율 이상이어야 "뾰족".
// 올리면(예 0.25) 더 뾰족한 것만, 내리면(예 0.05) 납작해도 통과.
뾰족비율 = 0.15;
높이1 = SA(1) - SB(1);
높이2 = SA(2) - SB(2);
최대높이 = max(max(높이1, 높이2), 0.0001);
뾰족2 = (abs(높이1 - 높이2) / 최대높이) >= 뾰족비율;

// ---- [왼쪽구름층] 삼각형 왼쪽 8봉 중 반대색 구름(SB>SA) 봉수 ----------
// 폭1 은 SA(2)~SA(9), 폭2 는 SA(3)~SA(10) 구간을 본다. 구름이 있는 봉마다 1점.
// iff 가 안 되는 환경이면 iff(조건,1,0) 대신 (조건) 을 그대로 더해도 됨.
왼쪽구름1 = iff(SB(2) > SA(2), 1, 0) + iff(SB(3) > SA(3), 1, 0) + iff(SB(4) > SA(4), 1, 0) + iff(SB(5) > SA(5), 1, 0) + iff(SB(6) > SA(6), 1, 0) + iff(SB(7) > SA(7), 1, 0) + iff(SB(8) > SA(8), 1, 0) + iff(SB(9) > SA(9), 1, 0);
왼쪽구름2 = iff(SB(3) > SA(3), 1, 0) + iff(SB(4) > SA(4), 1, 0) + iff(SB(5) > SA(5), 1, 0) + iff(SB(6) > SA(6), 1, 0) + iff(SB(7) > SA(7), 1, 0) + iff(SB(8) > SA(8), 1, 0) + iff(SB(9) > SA(9), 1, 0) + iff(SB(10) > SA(10), 1, 0);
// 8봉 중 몇 봉 이상 구름이 있어야 하는지. 올리면(예 6) 엄격, 내리면(예 3) 느슨.
왼쪽구름최소봉수 = 4;

// ---- 조합: 폭1 / 폭2 보라 신호 --------------------------------------
// 폭1: 왼쪽 평탄 2봉(FlatB2) + 삼각형1 + 왼쪽구름 8봉 중 4봉 이상
// 폭2: 왼쪽 평탄 2봉(FlatB3) + 삼각형2 + 뾰족 + 왼쪽구름 8봉 중 4봉 이상
// (기존 검색식의 폭3 삼각형 Bull_3 은 스크립트에 없는 형태라 제외했음)
보라1 = FlatB2 && 삼각1 && (왼쪽구름1 >= 왼쪽구름최소봉수);
보라2 = FlatB3 && 삼각2 && 뾰족2 && (왼쪽구름2 >= 왼쪽구름최소봉수);
보라삼각형 = 보라1 || 보라2;
