// 흰색 배지 기준 검색식 (상승 삼각형) - 스크립트 v5.28 흰색 기준과 동일
// 사용 함수: highest, lowest, max, abs, 사칙연산, 비교, &&, || 만 사용
// 흰색은 삼각형 완성(신호봉) 뒤 2봉째, 즉 현재봉이 신호봉보다 2봉 뒤일 때 신호가 난다
// 그래서 삼각형 관련 봉 번호는 보라 검색식보다 모두 2씩 뒤로 밀려 있다
// (신호봉 = 2봉 전, 오른쪽 기준봉)
// [삼각형] 신호봉 기준 폭1은 SA가 SB 위로 1봉만 올라갔다 내려옴, 폭2는 2봉
// [왼쪽평탄구간] 삼각형 왼쪽 SB 불변 3봉 이상 (왼쪽 기준봉 포함)
// [오른쪽평탄구간] 신호봉부터 현재봉까지 3봉 동안 SB 동일
// [왼쪽구름층] 삼각형 왼쪽 8봉 중 반대색 구름(SB가 SA보다 위)이 6봉 이상 (두께는 안 봄)
// [오른쪽구름층] 신호봉부터 현재봉까지 3봉 모두 반대색 구름이 있음 (두께는 안 봄)
// [등록시점] 신호봉 뒤 2봉째인 현재봉에서 신호
// ================================================================

// [왼쪽평탄구간]+[오른쪽평탄구간] SB 불변 봉수
// 폭1은 현재봉부터 6봉 전까지 SB 동일 (오른쪽 3봉 + 삼각형 1봉 + 왼쪽 3봉 - 겹침 포함, 총 7봉)
// 폭2는 현재봉부터 7봉 전까지 SB 동일 (총 8봉)
// 왼쪽 평탄 최소 봉수를 바꾸려면 아래 Flat 식의 봉 번호를 늘리거나 줄인다
Flat3 = (SB(1) == SB) && (SB(2) == SB(1));
Flat7 = Flat3 && (SB(3) == SB(2)) && (SB(4) == SB(3)) && (SB(5) == SB(4)) && (SB(6) == SB(5));
Flat8 = Flat7 && (SB(7) == SB(6));

// [삼각형] 신호봉(2봉 전) 기준 SA가 SB 위로 올라갔다 내려온 모양
// 폭1은 SA(4)가 왼쪽 기준봉, SA(3)이 삼각형, SA(2)가 오른쪽 기준봉(신호봉)
// 폭2는 SA(5)가 왼쪽 기준봉, SA(4)와 SA(3)이 삼각형 2봉, SA(2)가 오른쪽 기준봉(신호봉)
Tri1 = (SA(4) <= SB(4)) && (SA(3) > SB(3)) && (SA(2) <= SB(2));
Tri2 = (SA(5) <= SB(5)) && (SA(4) > SB(4)) && (SA(3) > SB(3)) && (SA(2) <= SB(2));

// [삼각형] 꼭짓점이 뾰족한지 (폭2에만 적용, 폭1은 항상 통과)
// 올리면(예 0.25) 더 뾰족한 것만, 내리면(예 0.05) 납작해도 통과
SharpRatio = 0.15;
Hgt1 = SA(3) - SB(3);
Hgt2 = SA(4) - SB(4);
HgtMax = max(max(Hgt1, Hgt2), 0.0001);
Sharp2 = (abs(Hgt1 - Hgt2) / HgtMax) >= SharpRatio;

// [왼쪽구름층]+[오른쪽구름층] 봉마다 반대색 구름(SB가 SA보다 위)이면 1, 아니면 0
// (SB-SA) 를 그 절대값(최소 0.0001)으로 나누면 양수 1, 음수 -1, 0이면 0 이 되고
// max(값, 0) 으로 구름 있는 봉만 1이 된다
Cl0 = max((SB - SA) / max(abs(SB - SA), 0.0001), 0);
Cl1 = max((SB(1) - SA(1)) / max(abs(SB(1) - SA(1)), 0.0001), 0);
Cl2 = max((SB(2) - SA(2)) / max(abs(SB(2) - SA(2)), 0.0001), 0);
Cl3 = max((SB(3) - SA(3)) / max(abs(SB(3) - SA(3)), 0.0001), 0);
Cl4 = max((SB(4) - SA(4)) / max(abs(SB(4) - SA(4)), 0.0001), 0);
Cl5 = max((SB(5) - SA(5)) / max(abs(SB(5) - SA(5)), 0.0001), 0);
Cl6 = max((SB(6) - SA(6)) / max(abs(SB(6) - SA(6)), 0.0001), 0);
Cl7 = max((SB(7) - SA(7)) / max(abs(SB(7) - SA(7)), 0.0001), 0);
Cl8 = max((SB(8) - SA(8)) / max(abs(SB(8) - SA(8)), 0.0001), 0);
Cl9 = max((SB(9) - SA(9)) / max(abs(SB(9) - SA(9)), 0.0001), 0);
Cl10 = max((SB(10) - SA(10)) / max(abs(SB(10) - SA(10)), 0.0001), 0);
Cl11 = max((SB(11) - SA(11)) / max(abs(SB(11) - SA(11)), 0.0001), 0);
Cl12 = max((SB(12) - SA(12)) / max(abs(SB(12) - SA(12)), 0.0001), 0);
// [오른쪽구름층] 현재봉, 1봉 전, 신호봉(2봉 전) 모두 구름이 있어야 함 (3이면 통과)
RightCl = Cl0 + Cl1 + Cl2;
// [왼쪽구름층] 폭1은 삼각형 왼쪽 8봉이 4~11봉 전, 폭2는 5~12봉 전
LeftCl1 = Cl4 + Cl5 + Cl6 + Cl7 + Cl8 + Cl9 + Cl10 + Cl11;
LeftCl2 = Cl5 + Cl6 + Cl7 + Cl8 + Cl9 + Cl10 + Cl11 + Cl12;
// 8봉 중 몇 봉 이상 구름이 있어야 하는지, 올리면(예 7) 엄격, 내리면(예 5) 느슨
LeftClMin = 6;

// 조합
// 폭1은 왼쪽 평탄 3봉 + 오른쪽 평탄 3봉(Flat7) + 삼각형 + 왼쪽구름 6봉 이상 + 오른쪽구름 3봉
// 폭2는 같은 조건에 Flat8 + 뾰족

White1 = Flat7 && Tri1 && (LeftCl1 >= LeftClMin) && (RightCl >= 3);
