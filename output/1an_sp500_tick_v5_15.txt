function recordRealBarInterval(label, barEpochNow) {
    var prevEpoch = lastRealBarEpochByLabel[label];
    if (prevEpoch !== undefined && prevEpoch !== null && barEpochNow > prevEpoch) {
        if (!recentBarIntervals[label]) recentBarIntervals[label] = [];
        recentBarIntervals[label].push(barEpochNow - prevEpoch);
        if (recentBarIntervals[label].length > RECENT_BAR_INTERVAL_WINDOW) recentBarIntervals[label].shift();
    }
    lastRealBarEpochByLabel[label] = barEpochNow;
}
function recentAvgBarIntervalSec(label) {
    var arr = recentBarIntervals[label];
    if (!arr || arr.length < GATE_SANITY_MIN_SAMPLES) return null; // 기준 삼을 실측치 부족 - 판단 보류
    var sum = 0;
    for (var i = 0; i < arr.length; i++) sum += arr[i];
    return sum / arr.length;
}

// [신규][2026-09-09][틱봉 여유치 개편 - 시간대구분+주기별 차등]
// 예전엔 tickRate(스크립트 시작부터의 누적평균)만 쓰고, 여유치도 82
// 고정이었다 - 장 활발한 시간/한산한 시간을 구분 못 하고, 짧은 주기
// (틱이 자주 도는)와 긴 주기를 똑같이 취급하는 문제가 분봉에서 먼저
// 확인됐던 것과 동일 원인. 2안의 시간대구분(PEAK_HOUR_START/END)을
// 그대로 이식하고, 여유치를 주기별로 차등화: 300틱이하=+500,
// 600틱이하=+300, 600틱초과=+200.
var PEAK_HOUR_START = 20;
var PEAK_HOUR_END = 2;
function isPeakVolumeWindowKST() {
    var h = new Date().getHours();
    if (PEAK_HOUR_START <= PEAK_HOUR_END) return h >= PEAK_HOUR_START && h < PEAK_HOUR_END;
    return h >= PEAK_HOUR_START || h < PEAK_HOUR_END; // 자정걸침(예: 20~02시)
}
function tickCoverageFloor(cycle) {
    return cycle <= 300 ? 500 : (cycle <= 600 ? 300 : 200);
}
function computeAdaptiveRequestCount(cycle, fileLabel) {
    var originalMax = computeRequestCount(cycle);
    var info = lastCycleBarInfo[fileLabel];
    var nowEpoch = lastTickEpoch || Math.floor(Date.now() / 1000);
    var scriptElapsedSec = Math.max(1, (Date.now() - scriptStartTime) / 1000);
    // [수정] 누적평균 대신, 지금이 활발한 시간대인지에 따라 관측된
    // 누적평균을 참고치로만 쓰되, 활발/한산 여부로 최종 여유치를 가른다.
    var tickRate = totalTicksReceived / scriptElapsedSec;
    var baseFloor = tickCoverageFloor(cycle);
    var peakMultiplier = isPeakVolumeWindowKST() ? 1.5 : 1; // 활발한 시간대는 여유치를 1.5배로
    var coverageFloor = Math.ceil(baseFloor * peakMultiplier);
    var oldestEpoch = oldestPendingSignalEpoch(fileLabel);
    if (oldestEpoch !== null && tickRate > 0) {
        var barsSinceOldest = Math.ceil(Math.max(0, nowEpoch - oldestEpoch) * tickRate / cycle);
        coverageFloor = Math.max(coverageFloor, barsSinceOldest + baseFloor);
    }
    if (!info) return originalMax; // 첫 요청은 항상 최대치(전체이력 확보)
    var elapsedSec = Math.max(1, nowEpoch - info.newestEpoch);
    var expectedNewBars = Math.ceil(elapsedSec * tickRate / cycle);
    var adaptiveCount = Math.ceil(expectedNewBars * 1.5) + baseFloor;
    return Math.min(originalMax, Math.max(coverageFloor, adaptiveCount));
}

function pruneDedupStructures() {
    if (lastTickEpoch === null) return;
    var cutoff = lastTickEpoch - DEDUP_RETENTION_SEC;
    Object.keys(pretty2Written).forEach(function (k) { if (pretty2Written[k].epoch < cutoff) delete pretty2Written[k]; });
    var removedSeenKeys = 0, removedPriceEntries = 0, removedPriceKeys = 0, removedResolutions = 0, removedOrphaned = 0, removedRealtime = 0;
    Object.keys(seenKeys).forEach(function (k) { if (seenKeys[k] < cutoff) { delete seenKeys[k]; removedSeenKeys++; } });
    Object.keys(seenKeysByPrice).forEach(function (pk) {
        var list = seenKeysByPrice[pk];
        var kept = list.filter(function (entry) { return entry.epoch >= cutoff; });
        removedPriceEntries += (list.length - kept.length);
        if (kept.length === 0) { delete seenKeysByPrice[pk]; removedPriceKeys++; } else seenKeysByPrice[pk] = kept;
    });
    Object.keys(recentResolutions).forEach(function (k) { if (recentResolutions[k].epoch < cutoff) { delete recentResolutions[k]; removedResolutions++; } });
    Object.keys(orphanedSignalDetails).forEach(function (k) { if (orphanedSignalDetails[k].epoch < cutoff) { delete orphanedSignalDetails[k]; removedOrphaned++; } });
    Object.keys(lastSeenRealTime).forEach(function (k) { if (lastSeenRealTime[k].epoch < cutoff) { delete lastSeenRealTime[k]; removedRealtime++; } });
    if (removedSeenKeys || removedPriceEntries || removedPriceKeys || removedResolutions || removedOrphaned || removedRealtime) {
        logDiagnostic("[DIAG][DEDUP_PRUNE] seenKeys=-" + removedSeenKeys + " priceEntries=-" + removedPriceEntries + " priceKeys=-" + removedPriceKeys + " resolutions=-" + removedResolutions + " orphaned=-" + removedOrphaned + " realtime=-" + removedRealtime);
        pendingStateDirty = true;
    }
}

var fileWriteBuffer = [], FILE_WRITE_BUFFER_MAX = 2000;
function writeLinesToFile(lines) {
    var allLines = fileWriteBuffer.concat(lines || []);
    if (allLines.length === 0) return;
    try { timedPrintOnFile(OUTPUT_FILE, allLines.join("\n") + "\n"); fileWriteBuffer = []; }
    catch (e) { if (allLines.length > FILE_WRITE_BUFFER_MAX) allLines = allLines.slice(allLines.length - FILE_WRITE_BUFFER_MAX); fileWriteBuffer = allLines; }
}

function SlidingExtreme(size, mode) { this.size = size; this.mode = mode; this.buf = []; this.idx = 0; }
SlidingExtreme.prototype.push = function (val) {
    var i = this.idx++;
    var better = (this.mode === 'max') ? function (a, b) { return a <= b; } : function (a, b) { return a >= b; };
    while (this.buf.length && better(this.buf[this.buf.length - 1].v, val)) this.buf.pop();
    this.buf.push({ v: val, i: i });
    while (this.buf.length && this.buf[0].i <= i - this.size) this.buf.shift();
    return this.buf[0].v;
};

// (1)골(계곡)이 충분히 뾰족한지(펑퍼짐 걸러냄) (2)그 직전에 이미 반대편
// 언덕/골짜기가 있었는지(빈약한 이전구름 걸러냄) (3)회복 직후 구름이
// 계속 벌어지는 방향인지. saHist/sbHist는 index0=현재봉, index1=1봉전...
// 순서라(unshift로 쌓임) 사용자가 준 수식의 SA(N)/SB(N) 표기와 그대로
// 대응된다. isBull=true면 Bull(선행스팬A가 평탄, SB가 위에서 찌르고
// 회복), false면 Bear(선행스팬B가 평탄, SA가 아래서 찌르고 회복).
function computeAtrFromHist(saHist) {
    var atr = 0;
    for (var i = 0; i < 14 && i < saHist.length - 1; i++) atr += Math.abs(saHist[i] - saHist[i + 1]);
    return Math.max(atr / 14, 0.0001);
}
function checkPrettyTriangle(saHist, sbHist, isBull) {
    if (saHist.length < 8 || sbHist.length < 8) return false;
    var atr = computeAtrFromHist(saHist);
    var threshold = atr * PRETTY_TRIANGLE_RATIO;
    if (!isBull) {
        // Bear: 골 깊이(SB - 최근3봉 중 SA 최저치), 이전언덕(SA가 SB보다 위였던 5~7봉전)
        var golDepth = sbHist[0] - Math.min(saHist[0], saHist[1], saHist[2]);
        var prevHill = (saHist[5] > sbHist[5]) || (saHist[6] > sbHist[6]) || (saHist[7] > sbHist[7]);
        var expanding = (saHist[0] - sbHist[0]) > (saHist[1] - sbHist[1]);
        return golDepth >= threshold && prevHill && expanding;
    } else {
        // Bull: 골 깊이(최근3봉 중 SB 최고치 - SA), 이전골짜기(SB가 SA보다 아래였던 5~7봉전)
        var golDepthBull = Math.max(sbHist[0], sbHist[1], sbHist[2]) - saHist[0];
        var prevValley = (sbHist[5] < saHist[5]) || (sbHist[6] < saHist[6]) || (sbHist[7] < saHist[7]);
        var expandingBull = (saHist[0] - sbHist[0]) > (saHist[1] - sbHist[1]);
        return golDepthBull >= threshold && prevValley && expandingBull;
    }
}
var PRETTY_TRIANGLE_RATIO = 0.15; // 실측하며 조절할 값(높을수록 더 뾰족한 것만 통과)
// [신규][2026-09-26][흰다이아몬드 배지 - 오른쪽 구름대 확인] 기존
// 다이아몬드(checkPrettyTriangle)는 그대로 두고, "삼각형 완성 후 반대편에도

// 호출되는 이 SignalCalculator.onBar()의 saHist/sbHist 누적을 재사용해서

var PRETTY_TRIANGLE2_LOOKAHEAD_BARS = 8; // checkPrettyTriangle의 왼쪽 8봉 대칭
// [수정][2026-09-26][v5.8 - 오른쪽 구름대 판정을 독립된 케이스들로 분리]
// 사용자 지적: 얇아도 평탄구간이 넓고 삼각형이 예쁘면(=기존 다이아몬드를

// 탈락시킴 - 그리고 실제 차트엔 넓은 평탄구간 안에서 잠깐 한 번
// 아래로/위로 삐끗(V자)했다가 회복하는 형태도 있음. 사용자 결정: 여러
// 기준을 하나로 합치지 말고 "케이스"로 서로 완전히 독립시켜서, 나중에
// 특정 케이스만 골라 끄거나 되돌릴 수 있게 하고, 통과한 케이스 번호를
// 그대로 기록해 배지 마우스오버에 "몇 번 케이스로 통과했는지" 보여준다
// (중복 통과시 "1,2"처럼 여러 개). 각 케이스는 서로를 몰라도 되는 순수
// 함수 - 하나를 고치거나 빼도 나머지에 영향 없음.
function checkRightCloudCase1_ThickAverage(saHist, sbHist, isBull) {
    // 케이스1(두께평균 - 가장 엄격/확실): 한 봉이라도 두께가 0 이하(구름
    // 소멸/역전)면 무조건 탈락(하드 플로어), 나머지는 N봉 평균 두께가
    // threshold 이상이면 통과 - v5.7 로직 그대로, 케이스1로 이름만 분리.
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    var atr = computeAtrFromHist(saHist);
    var threshold = atr * PRETTY_TRIANGLE_RATIO;
    var sumThickness = 0;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) return false;
        sumThickness += thickness;
    }
    return (sumThickness / PRETTY_TRIANGLE2_LOOKAHEAD_BARS) >= threshold;
}
function checkRightCloudCase2_ThinButPresent(saHist, sbHist, isBull) {
    // 케이스2(얇아도 유지): 두께 크기(threshold)는 아예 안 보고, N봉 내내
    // 단 한 번도 역전/소멸하지 않았는지만 확인 - 케이스1보다 완화된 조건
    // (케이스1을 통과하면 이 케이스도 항상 자동으로 통과하는 상위개념).
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) return false;
    }
    return true;
}
var PRETTY_TRIANGLE2_DIP_TOLERANCE_BARS = 2; // N=8봉 중 최대 허용 역전/소멸 봉수(실측하며 조절할 값)
function checkRightCloudCase3_BriefDipTolerant(saHist, sbHist, isBull) {
    // 케이스3(짧은 V자 역전 허용): N봉 중 최대 DIP_TOLERANCE개 봉까지는
    // 역전/소멸돼도 되고, 나머지(절대다수)만 정상 방향이면 통과 - 넓은
    // 평탄구간 안에서 잠깐 한 번 삐끗했다 회복하는 형태까지 인식(케이스2를
    // 통과하면 이 케이스도 항상 자동으로 통과하는 상위개념).
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    var invertedCount = 0;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) invertedCount++;
    }
    return invertedCount <= PRETTY_TRIANGLE2_DIP_TOLERANCE_BARS;
}
// 세 케이스를 서로 독립적으로 평가해 통과한 케이스 번호만 모아 반환 -
// 어느 하나라도 통과하면 흰다이아몬드 확정(matched=true). cases 배열은
// 그대로 신호파일/대시보드까지 실어보내 배지 마우스오버에 표시한다.
function evalRightCloudCases(saHist, sbHist, isBull) {
    var cases = [];
    if (checkRightCloudCase1_ThickAverage(saHist, sbHist, isBull)) cases.push(1);
    if (checkRightCloudCase2_ThinButPresent(saHist, sbHist, isBull)) cases.push(2);
    if (checkRightCloudCase3_BriefDipTolerant(saHist, sbHist, isBull)) cases.push(3);
    return { matched: cases.length > 0, cases: cases };
}

function nowServerTimeStr() {

    // 예전엔 marketDataObj.date/.time(예스트레이더 실시간시세 필드)을

    // UTC 기준시각에서 미국 서머타임 규칙(3월 둘째주 일요일~11월 첫째주

    // 시간대 설정과 무관하게 항상 정확한 시카고(CT) 현재시각을 반환하도록
    // 근본적으로 재설계했다(2안/2+1안과 동일 로직).
    var nowUtc = new Date();

    var marchFirst = new Date(Date.UTC(year, 2, 1));
    var secondSundayMarch = 1 + ((7 - marchFirst.getUTCDay()) % 7) + 7;
    var dstStart = new Date(Date.UTC(year, 2, secondSundayMarch, 8, 0, 0));
    var novFirst = new Date(Date.UTC(year, 10, 1));
    var firstSundayNov = 1 + ((7 - novFirst.getUTCDay()) % 7);
    var dstEnd = new Date(Date.UTC(year, 10, firstSundayNov, 7, 0, 0));
    var chicagoOffsetHours = (nowUtc >= dstStart && nowUtc < dstEnd) ? -5 : -6;
    var chicagoMs = nowUtc.getTime() + chicagoOffsetHours * 3600 * 1000;
    var cd = new Date(chicagoMs);
    var pad = function (n) { return ("0" + n).slice(-2); };
    return cd.getUTCFullYear() + ("0" + (cd.getUTCMonth() + 1)).slice(-2) + ("0" + cd.getUTCDate()).slice(-2) + " " +
        pad(cd.getUTCHours()) + ":" + pad(cd.getUTCMinutes()) + ":" + pad(cd.getUTCSeconds());
}

function countBarsAfterEpoch(bars, epochRef) {
    if (epochRef === undefined || epochRef === null) return 0;
    var firstMatchIdx = -1;
    for (var i = 0; i < bars.length; i++) {
        if (dateTimeToEpochApprox(String(bars[i].sdate), bars[i].stime) >= epochRef) { firstMatchIdx = i; break; }
    }
    if (firstMatchIdx === -1) return 0;
    var firstMatchEpoch = dateTimeToEpochApprox(String(bars[firstMatchIdx].sdate), bars[firstMatchIdx].stime);
    var startCountIdx = (firstMatchEpoch === epochRef) ? firstMatchIdx + 1 : firstMatchIdx;
    return Math.max(0, bars.length - startCountIdx);
}

function runPendingResolutionPass(filterFn, resolveFn, siteName, markDirtyOnKeep) {
    var stillPending = [], resolvedLines = [];
    function keepPending(sig) { stillPending.push(sig); if (markDirtyOnKeep) excelMarkDirty(sig); }
    var beforeSnapshot = pendingSignals.slice();
    var explainedPks = {};
    for (var i = 0; i < pendingSignals.length; i++) {
        var sig = pendingSignals[i];
        if (!filterFn(sig)) { stillPending.push(sig); continue; }
        var resolved;
        try {
            resolved = resolveFn(sig, resolvedLines, function () { explainedPks[pkOf(sig)] = true; });
        } catch (eResolve) {
            // [근본수정][2026-09-25][격리실패 - 신호1개 예외가 배열 전체를 멈추던
            // 버그] 이 for-loop는 신호별 try/catch가 없어서, 배열 앞쪽 신호
            // 하나가 resolveFn(applyPriceToSignal 등) 처리중 예외를 던지면
            // 그 뒤에 있는 모든 신호가 이번 패스에서 통째로 스킵됐다(다음
            // 틱/사이클에서도 배열 순서가 그대로라 매번 반복 - 영구정지).
            // 게다가 이 함수의 호출부(updatePendingSignals)는 다시
            // Main_OnUpdateMarket의 try{}catch(e){}에 감싸여 있어 예외가
            // 완전히 조용히 삼켜졌다 - 로그에 아무 흔적도 안 남는 이유.
            // 실제 재현: T0340/T0345류 신호가 26봉 게이트 초과+가격돌파
            // 확인됐는데도 감시중에서 영원히 안 풀리는 증상을 Node.js
            // 샌드박스에서 원본함수 그대로 실행해 재현/증명함(고의로 손상시킨

            // try/catch를 격리해서, 하나가 깨져도 그 신호만 "이번엔 미해결"
            // 처리하고 나머지는 계속 정상 판정되게 한다. 동시에 어떤 신호의
            // 어떤 필드가 왜 깨졌는지 진단로그에 바로 남겨서(전에는 수동으로
            // 대시보드를 뒤져야 했음) 다음 발생시 즉시 원인 확인 가능하게 한다.
            resolved = false;
            try {
                var pkErr = pkOf(sig);
                var errMsg = (eResolve && eResolve.message) ? eResolve.message : String(eResolve);
                var errStack = (eResolve && eResolve.stack) ? String(eResolve.stack).replace(/\n/g, " | ").slice(0, 400) : "";
                logDiagnostic("[DIAG][RESOLVE_EXCEPTION] site=" + siteName + " pk=" + pkErr + " state=" + sig.state + " gate=" + (sig.barsElapsedSinceRegistration || 0) + "/" + REQUIRED_BAR_DELAY + " err=" + errMsg + " stack=" + errStack);

            } catch (eLog) {}
        }
        if (!resolved) keepPending(sig);
    }
    pendingSignals = stillPending;
    auditPendingReassignment(beforeSnapshot, pendingSignals, explainedPks, siteName);
    if (resolvedLines.length > 0) { writeLinesToFile(resolvedLines); pendingStateDirty = true; }
}

function formatHistoryWhen(whenStr) { if (typeof whenStr !== "string") return whenStr; var m = whenStr.match(/^(\d{4})(\d{2})(\d{2}) (.+)$/); return m ? (m[2] + "/" + m[3] + " " + m[4]) : whenStr; }
// [2026-08-17 이식][2안 틱봉 2026-08-13 근본원인수정을 1안에도 동일 적용]
// 예전엔 무조건 배열 끝에 push만 해서, "언제 pushStateHistory가
// 호출됐는지(wall-clock 호출순서)"와 "그 항목이 실제로 가리키는
// 시각(embedded epoch)"이 다르면 순서가 뒤섞였다. 이제 매번 끝에 push하는
// 대신, 각 항목의 실제 시각(epoch)을 기준으로 항상 올바른 시간순 위치에
// 삽입한다 - 같은 epoch끼리는 기존 호출순서를 그대로 유지(안정정렬).
function pushStateHistory(sig, whenStr, stateWord, price, extraDetail) {
    if (!sig.stateHistory) sig.stateHistory = [];
    var dw = formatHistoryWhen(whenStr);
    var epoch = null;
    if (typeof whenStr === "string") {
        var wsParts = whenStr.split(" ");
        if (wsParts.length === 2 && wsParts[0].length === 8) { try { epoch = dateTimeToEpochApprox(wsParts[0], wsParts[1]); } catch (e) { epoch = null; } }
    }
    var entry = { t: dw, s: stateWord + "(" + dw + ")@" + Number(price).toFixed(2) + (extraDetail ? "[" + extraDetail + "]" : ""), epoch: epoch };
    if (epoch === null || sig.stateHistory.length === 0) { sig.stateHistory.push(entry); return entry; }
    var insertAt = sig.stateHistory.length;
    for (var i = sig.stateHistory.length - 1; i >= 0; i--) {
        var existingEpoch = sig.stateHistory[i].epoch;
        if (existingEpoch === null || existingEpoch === undefined || existingEpoch <= epoch) { insertAt = i + 1; break; }
        insertAt = i;
    }
    sig.stateHistory.splice(insertAt, 0, entry);
    return entry;
}
function buildResolvedReasonField(sig, result, whenStr, price, cellTextOverride) {
    var shortReason;
    if (cellTextOverride) shortReason = cellTextOverride;
    else if (result === "ACHIEVED") shortReason = "ACHIEVED(" + whenStr + ")@" + Number(price).toFixed(2);
    else if (result === "INVALID") {
        var rr = sig.resolveReason || "";
        if (rr.indexOf("NO_PULLBACK") === 0) shortReason = "NO_PULLBACK_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else if (rr.indexOf("NO_REBOUND") === 0) shortReason = "NO_REBOUND_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else if (rr.indexOf("AMBIGUOUS_BAR") === 0) shortReason = "AMBIGUOUS_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else shortReason = "INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
    } else shortReason = String(result);
    var histStr = (sig.stateHistory && sig.stateHistory.length > 0) ? " / HISTORY:" + sig.stateHistory.map(function (h) { return h.s; }).join(" -> ") : "";
    return (shortReason + histStr).replace(/,/g, ";");
}
function applyPriceToSignal(sig, curPrice, curDateStr, curTimeStr, gateOpen, callerTag) {

    if (gateOpen === true && !isGateCountSane(sig, curDateStr, curTimeStr)) {

    }

            else if (sig.state === "PULLBACK" && curPrice > sig.price) { sig.state = "WAITING"; } // [수정][2026-09-29] 게이트 닫힘 구간의 눌림<->대기 왕복은 이력을 무한히 늘리므로 상태값만 바꾸고 이력은 남기지 않는다(눌림확인은 신호당 첫 1회만 기록)
        } else {

            else if (sig.state === "PULLBACK" && curPrice < sig.price) { sig.state = "WAITING"; }

    }
    if (sig.isBull) {
        if (sig.state === "WAITING") {
            if (curPrice < sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); }
            else if (curPrice > sig.price) { sig.resolveReason = "NO_PULLBACK->INVALID(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "INVALID_CONFIRMED", curPrice, gateDetail); return "INVALID"; }
        } else if (sig.state === "PULLBACK") {
            if (curPrice >= sig.price) { sig.resolveReason = "PULLBACK->ACHIEVED(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "ACHIEVED_CONFIRMED", curPrice, gateDetail); return "ACHIEVED"; }
        }
    } else {
        if (sig.state === "WAITING") {
            if (curPrice > sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); }

        }
    }
    return null;
}

            logPendingRemoval(sig, result, reasonField); // [근본수정][2026-09-26][v5.3] 예전엔 짧은 태그("REALTIME_RESOLVE@...")만 REMOVAL_LOG에 남겨서, 1차 STATUS_ACHIEVED/INVALID 매칭이 실패해 REMOVAL_LOG 복구경로로 대체될 때 warmupResolve 경로(rReasonField 그대로 전달)와 달리 실시간 경로만 사유가 텅 비어 보이는 비대칭이 있었다(사용자 지적: "여러 곳에서 판정하는데 한 곳만 제대로") - 이미 계산해둔 reasonField(전체 HISTORY 포함)를 그대로 넘겨 두 경로를 동일하게 맞춘다.
            markExplained();
            return true;
        },
        "realtimeResolve",
        false // markDirtyOnKeep: [수정][2026-09-29] 틱마다 전체 표시하던 것을 끄고, 실제 변경은 위 resolveFn에서 개별 표시
    );
}
function findFirstBarIndexAtOrAfterEpoch(bars, epoch) {
    var lo = 0, hi = bars.length;
    while (lo < hi) {
        var mid = (lo + hi) >> 1;
        var midEpoch = dateTimeToEpochApprox(String(bars[mid].sdate), bars[mid].stime);
        if (midEpoch < epoch) lo = mid + 1; else hi = mid;
    }
    return lo;
}
function resolveHistorically(sig, bars) {
    if (bars.length === 0) return null;
    var sigEpochCached = sig.signalEpoch;
    // [2026-09-25 이력] 카운팅커서(lastAppliedBarEpoch)/판정커서(lastJudgedBarEpoch)
    // 분리 자체는 유지한다(WAITING/PULLBACK 코스메틱 이력 중복방지 목적).
    // [근본수정][2026-09-29][WARM 안전망 영구봉쇄 버그] 그런데 그 분리 이후에도
    // 이 판정커서를 "달성/무효(터미널) 판정 스킵" 기준으로 그대로 썼다 - 그런데
    // processLiveBar(실시간 봉완결)는 이 커서를 매 완결봉마다 "지금"까지
    // 무조건 전진시킨다(가격판정 없이, 카운팅커서와 함께). 그 결과 실시간
    // 추적이 살아있는 신호는(항상 그렇다 - REALTIME_TRACK_ENABLED 상시 true)
    // 이 커서가 늘 "방금"까지 따라붙어서, WARM이 실제 차트를 재조회해도
    // startIndex가 배열 끝을 넘어가 이 for문이 통째로 한 번도 안 돈다.
    // 실시간 판정(updatePendingSignals)은 그 순간 들어온 틱 가격 1개만 보는데,
    // 그 틱이 하필 눌림반등/달성 문턱을 놓치면(체결이 뜸한 시간대) - 그 구간의
    // "진짜 high/low"를 재확인할 기회가 영구히 사라진다(NASDAQ100 분봉
    // M0016/M0017 실사례를 정확한 조건으로 재현 완료 - 등록가를 한참 뚫고
    // 내려간 실제 차트 데이터를 갖고 있어도 달성처리가 전혀 안 됨. 틱봉도
    // 동일 구조이므로 같이 고친다).
    // 판정커서는 "WAITING/PULLBACK 코스메틱 토글 재생만" 막는 용도로 한정하고,
    // ACHIEVED/INVALID 확정 검사는 매 WARM 재조회마다 signalEpoch부터 매번
    // 그 봉의 실제 high/low로 다시 확인한다 - 이미 터미널 처리된 신호는
    // pendingSignals에서 빠져있어 재확인 대상이 아니므로(멱등), 중복확정
    // 위험이 없다. 코스메틱 이력 중복방지(원래 09-25/26 수정의 진짜 목적)는
    // alreadyCosmeticJudged 플래그로 그대로 보존한다(재현테스트로 중복 재발
    // 없음 확인).
    var judgedFromEpoch = (sig.lastJudgedBarEpoch !== undefined && sig.lastJudgedBarEpoch !== null) ? sig.lastJudgedBarEpoch : sigEpochCached;
    var countedFromEpoch = (sig.lastAppliedBarEpoch !== undefined && sig.lastAppliedBarEpoch !== null) ? sig.lastAppliedBarEpoch : sigEpochCached;
    var startBarIndex = findFirstBarIndexAtOrAfterEpoch(bars, sigEpochCached);
    var barsSinceSignal = sig.barsElapsedSinceRegistration || 0;
    // [근본수정][2026-09-29][2차][stale 게이트값을 과거봉에 잘못 적용하던 버그
    // - 재현테스트로 확인] barsSinceSignal은 여러 재조회에 걸쳐 누적되는
    // "영속" 카운터라, 위에서 항상 signalEpoch부터 다시 스캔하게 바꾼
    // 뒤로는 실시간이 이미 훨씬 뒤(미래) 봉까지 카운트를 올려놓은 상태에서
    // "과거의" 봉을 재검사할 때도 이 부풀려진 최종값을 그대로 써버려,
    // 실제로는 게이트가 열리기 한참 전인 초기 봉이 "게이트 열림" 판정을
    // 받는 조기확정 위험이 있었다(재현 확인 - gate=60으로 부풀려진 상태에서
    // 2번째 봉이 즉시 AMBIGUOUS_BAR로 확정되는 것까지 실제 재현). 틱봉은
    // 시간조건이 따로 없어서(gateOpen이 봉개수 하나로만 결정) 이 문제에
    // 특히 취약함을 확인 - trueBarCountAsOfThisBar는 이번 호출에서
    // signalEpoch부터 매번 처음부터 다시 세는 "이 봉 시점 기준" 카운트로,
    // 영속 카운터(barsSinceSignal, 대시보드 표시/실시간 연계용, 용도 그대로
    // 유지)와 분리해서 게이트 열림 판정에만 쓴다.
    var trueBarCountAsOfThisBar = 0;
    for (var bj = startBarIndex; bj < bars.length; bj++) {
        var bar = bars[bj];
        var barEpoch = dateTimeToEpochApprox(String(bar.sdate), bar.stime);
        if (barEpoch <= sigEpochCached) continue;
        // [2026-09-25 근본수정][filled(가짜/보간)봉이 게이트+판정에 섞이던 버그]
        // bar.filled=true는 재조회 응답에 구멍(누락)이 생겨 직전 정상봉의
        // 고가/저가를 그대로 복사해 채운 "실제 관측이 아닌" 값이다 - 다수결/
        // 이력 검증(barCheckHistory) 쪽은 이미 filled를 제외하고 있었는데
        // (cbar.filled continue), 정작 여기(26봉 게이트 카운트 및 달성/무효
        // 판정)는 필터링이 없어서 가짜봉도 그대로 카운트되고 그 가짜 고가/저가로
        // 달성/무효가 확정될 수 있었다(분봉 파일과 동일 재현 테스트로 확인).
        // 실제 새 데이터가 아니므로 게이트 카운트도, 가격판정도 전부 건너뛴다 -
        // 커서도 안 움직이므로 다음 재조회에서 이 구간이 실제 데이터로 채워지면
        // 그때 다시 정상 처리된다.
        if (bar.filled) continue;
        trueBarCountAsOfThisBar++;
        var alreadyCosmeticJudged = (barEpoch <= judgedFromEpoch); // [2026-09-29] 코스메틱(WAITING/PULLBACK 토글) 이력 중복방지 전용 플래그
        if (barEpoch > countedFromEpoch) {
            barsSinceSignal++;
            sig.barsElapsedSinceRegistration = barsSinceSignal;
            sig.lastAppliedBarEpoch = barEpoch;
            countedFromEpoch = barEpoch;
        }
        if (barEpoch > judgedFromEpoch) { sig.lastJudgedBarEpoch = barEpoch; judgedFromEpoch = barEpoch; }
        var gateOpen = (trueBarCountAsOfThisBar >= REQUIRED_BAR_DELAY); // [2026-09-29] 영속카운터 대신 이 봉 시점 기준 카운트 사용
        var firstPrice, secondPrice;
        if (sig.isBull) { firstPrice = bar.low; secondPrice = bar.high; } else { firstPrice = bar.high; secondPrice = bar.low; }
        if (!gateOpen) {
            // [2026-09-29] 게이트가 아직 닫혀있는 구간의 WAITING/PULLBACK
            // 코스메틱 토글은, 실시간이 이미 훑은 구간(alreadyCosmeticJudged)이면
            // 재생하지 않는다 - 09-25/26 수정의 원래 목적(중복이력 방지) 그대로 보존.
            if (!alreadyCosmeticJudged) { applyPriceToSignal(sig, firstPrice, bar.sdate, bar.stime, false, "resolveHistorically(WARM)"); applyPriceToSignal(sig, secondPrice, bar.sdate, bar.stime, false, "resolveHistorically(WARM)"); }
            continue;
        }
        if (sig.state === "WAITING" && bar.low < sig.price && bar.high > sig.price) {
            sig.resolveReason = "AMBIGUOUS_BAR(" + bar.sdate + "_" + bar.stime + ")->INVALID";
            pushStateHistory(sig, bar.sdate + " " + bar.stime, "INVALID_CONFIRMED", sig.price);
            return { result: "INVALID", sdate: bar.sdate, stime: bar.stime };
        }
        var stateBefore = sig.state;
        // [2026-09-14][죽은코드 제거] 확인단계(confirmHistoricalResult)가 항상
        // 즉시통과라, 스냅샷/롤백 없이 결과가 나오면 그대로 반환한다.
        var r1 = applyPriceToSignal(sig, firstPrice, bar.sdate, bar.stime, true, "resolveHistorically(WARM)");
        if (r1) return { result: r1, sdate: bar.sdate, stime: bar.stime };
        if (stateBefore === "WAITING" && sig.state === "PULLBACK") continue;
        var r2 = applyPriceToSignal(sig, secondPrice, bar.sdate, bar.stime, true, "resolveHistorically(WARM)");
        if (r2) return { result: r2, sdate: bar.sdate, stime: bar.stime };
    }
    return null;
}
var TICK_TRUE_ORIGIN_TOLERANCE_SEC = 10;
function isDuplicatePendingSignal(itemName, tf, kind, sdate, stime, price, trueSdate, trueStime) {
    // [근본수정][2026-09-09][진짜 원본시각(trueSdate/trueStime) 기준으로
    // 판정] 예전엔 되짚기 적용된 표시용 sdate/stime을 시간허용오차로
    // 비교했는데, 되짚기 계산이 재조회마다 흔들릴 수 있어 진짜 같은
    // 사건을 다른 사건으로 오판하는 문제가 분봉에서 실제사례로 확인됨.
    // [근본수정][2026-09-10][틱봉 전용: 완전일치->10초 허용오차 + 가격체크
    // 추가] 위 수정은 "분봉은 원본시각이 완전히 고정된다"는 전제였는데,
    // 틱봉은 다르다 - TickBarBuilder의 sdate/stime이 chartEx.GetSDate/
    // GetSTime에서 그대로 오는데, 이 원본 틱데이터 자체가 재조회마다
    // 살짝 달라지는 현상이 이미 예스스팟에 문의 접수된 상태(별도 확인).
    // 그래서 완전일치로는 몇 초 차이로 진짜 중복을 놓치는 사례가 실제
    // 화면에서 확인됨(T0305/T0310/T0315 각각 2번씩 등록, 5~8초 간격).
    // 36분 단위로 드리프트하던 옛날 버그(표시시각 기준)와는 크기가 전혀
    // 다른 문제라 10초 정도의 작은 허용오차로는 그 버그가 재발하지
    // 않는다(10초 << 36분). 가격(price) 체크도 이 분기엔 원래 없었어서
    // 같이 추가 - "정말 다른 두 신호가 우연히 몇 초 안에 겹치는" 잔여
    // 리스크까지 막는다.
    if (trueSdate && trueStime) {
        var thisTrueEpoch = dateTimeToEpochApprox(String(trueSdate), trueStime);
        for (var pj = 0; pj < pendingSignals.length; pj++) {
            var pp = pendingSignals[pj];
            if (pp.itemName === itemName && pp.tf === tf && pp.kind === kind &&
                pp.trueSdate && pp.trueStime && Math.abs(price - pp.price) < 0.01) {
                var ppTrueEpoch = dateTimeToEpochApprox(String(pp.trueSdate), pp.trueStime);
                if (Math.abs(thisTrueEpoch - ppTrueEpoch) <= TICK_TRUE_ORIGIN_TOLERANCE_SEC) return true;
            }
        }
        return false;
    }
    var thisEpoch = dateTimeToEpochApprox(String(sdate), stime);
    for (var pi = 0; pi < pendingSignals.length; pi++) {
        var p = pendingSignals[pi];
        if (p.itemName === itemName && p.tf === tf && p.kind === kind && Math.abs(p.price - price) < 0.01) {
            var pEpoch = (p.signalEpoch !== undefined && p.signalEpoch !== null) ? p.signalEpoch : dateTimeToEpochApprox(String(p.sdate), p.stime);
            if (Math.abs(thisEpoch - pEpoch) <= DUP_SIGNAL_TIME_TOLERANCE_SEC) return true;
        }
    }
    return false;
}
function findPendingMatch(itemName, tf, kind, sdate, stime, price) {
    var thisEpoch = dateTimeToEpochApprox(String(sdate), stime);
    for (var pi = 0; pi < pendingSignals.length; pi++) {
        var p = pendingSignals[pi];
        if (p.itemName === itemName && p.tf === tf && p.kind === kind && Math.abs(p.price - price) < 0.01) {
            var pEpoch = (p.signalEpoch !== undefined && p.signalEpoch !== null) ? p.signalEpoch : dateTimeToEpochApprox(String(p.sdate), p.stime);
            if (Math.abs(thisEpoch - pEpoch) <= DUP_SIGNAL_TIME_TOLERANCE_SEC) return p;
        }
    }
    return null;
}

// [근본수정: 단일 진실원] 신호 객체 생성 + REGISTERED 이력 기록을 이 함수
// 하나로 통일. trial(즉시판정) 경로와 registerPendingSignal(대기등록)

function buildNewSignalObject(itemName, tf, kind, sdate, stime, price, period, signalBarCount, initialBarsElapsed, registeredWhen, source, leftFlatStar, trueSdate, trueStime) {
    var regWhenForHistory = registeredWhen || (sdate + " " + stime);
    // [2026-09-02][등록경로 태그 보완] 2안과 동일 패턴 - 웜업(WARM)/실시간(REAL)
    // 등록경로를 stateHistory[0]에 즉시 남긴다.
    var sourceTag = source ? "[" + source + "]" : "";
    return {
        itemName: itemName, tf: tf, kind: kind, sdate: sdate, stime: stime, price: price,
        isBull: kind.indexOf("BULL") === 0, state: "WAITING", period: period,
        signalBarCount: signalBarCount, barsElapsedSinceRegistration: (initialBarsElapsed || 0),
        // [신규][2026-09-07][일관성 - 판정/철회에도 신호세기 기반 확인단계
        // 적용하려면 이 값이 신호 객체 자체에 남아있어야 한다](2안/2+1안과 동일)
        leftFlatStar: leftFlatStar || 0,
        lastAppliedBarEpoch: null,
        // [신규][2026-09-25][카운팅커서/판정커서 분리] resolveHistorically
        // 전용 판정커서 - null이면 signalEpoch부터 다시 판정을 시도한다
        // (기존 신호 복원시 이 필드가 없으면 자동으로 이 폴백이 걸려서,
        // 과거에 lastAppliedBarEpoch만으로 인해 재판정을 놓쳤던 신호도
        // 다음 재워밍업에서 처음부터 다시 판정된다).
        lastJudgedBarEpoch: null,
        // [긴급수정][2026-09-07][백데이팅과 게이트카운트 혼용 버그](2안/2+1안과
        // 동일 원인/동일 수정) 되짚기(sdate/stime, 화면표시/중복방지 전용)를
        // 게이트(signalEpoch) 계산에도 그대로 써버린 게 버그였다 - 반드시
        // trueSdate/trueStime(백데이팅 안 된 진짜 삼각형완성 시각)으로 계산.
        signalEpoch: dateTimeToEpochApprox(String(trueSdate || sdate), trueStime || stime), registeredAt: Date.now(),
        // [근본수정][2026-09-09][trueSdate/trueStime을 신호객체 필드로도
        // 저장] signalEpoch 계산에만 쓰고 정작 객체 자체엔 안 남기고
        // 있었다 - 그래서 isDuplicatePendingSignal이 이 값을 쓰도록
        // 고쳐도(sg.trueSdate 참조) 항상 undefined라 실질적으로 무효였음.
        trueSdate: trueSdate || sdate, trueStime: trueStime || stime,
        lastCheckedAt: null, lastCheckedCycle: null,
        lastCheckedReason: "1안:매사이클 전체재계산으로 추적(별도 실시간이벤트 없음)",
        stateHistory: [{ t: formatHistoryWhen(regWhenForHistory), s: "REGISTERED(" + formatHistoryWhen(regWhenForHistory) + ")" + sourceTag + "@" + Number(price).toFixed(2) }],
        _saveConfirmed: false
    };
}
function registerPendingSignal(itemName, tf, kind, sdate, stime, price, period, signalBarCount, initialBarsElapsed, source, leftFlatStar, trueSdate, trueStime, preWalkedSig) {
    if (isDuplicatePendingSignal(itemName, tf, kind, sdate, stime, price, trueSdate, trueStime)) { recordSuppressedDuplicate(itemName, tf, kind, sdate, stime, price, "DUP_AT_REGISTER", null); return false; }
    // [근본수정][2026-09-10][2안 v1.07~v1.09 검증패턴 적용] registeredWhen을

            liveCalcs[target.cycle] = new SignalCalculator();
            logDiagnostic("[DIAG][DEGRADED_COLDSTART] label=" + fileLabel + " 실시간계산기 콜드시딩(웜업복구 전까지 임시)");
        }
        degradedPeriods[fileLabel] = true;
        completedCycles[target.cycle] = true;
        logMsg("  !! [" + fileLabel + "] " + MAX_RETRY_ATTEMPTS_PER_PASS + "회 즉시재시도 전부 실패 - 이번 순환은 저품질(degraded), 다음 순환에서 재시도");
        // [신규][2026-09-25][대시보드 모달 알림] 화면(MessageLog)에만 남기던
        // 걸 신호파일에도 남겨서 bridge_signals.py/대시보드가 실시간으로
        // 알 수 있게 한다. 형식: item,fileLabel,DEGRADED_DEFERRED,date,time,cycle,재시도횟수
        try {

        } catch (eDeg) {}
    }
}

function processNext() {
    // [2026-09-02 근본수정][예스스탁 공식답변 반영 - 삭제/재요청 순서보장]
    // 삭제타이머(nEventID=5)가 요청 사이클과 완전히 독립적으로 자체
    // 재무장하도록 바꾼 뒤(2026-08-13, 삭제가 영원히 안 불릴 위험 방지),
    // 그 대가로 "다음 요청이 나가는 시점에 직전 차트객체가 아직 안
    // 지워진 채로 남아있을 수 있는" 새 위험이 생겼다 - 예스스탁 공식
    // 답변("삭제와 재요청 순서를 확실히 분리하라")과 정확히 일치하는
    // 문제로, 서로 다른 타임프레임에 동일한 가짜 가격의 신호가 동시에
    // 등록되는 사고로 재현 확인됨. 다음 요청을 보내기 직전에 매번 밀린
    // 삭제큐부터 먼저 확실히 비워서, "요청이 나갈 때는 항상 직전 객체가
    // 이미 지워진 상태"를 보장한다 - 원래 목적(삭제 무한지연 방지)은
    // 독립 타이머(nEventID=5)가 그대로 안전망으로 남아있어 훼손되지 않는다.
    if (pendingRemoveObjects.length > 0) {
        var objectsToRemoveNow = pendingRemoveObjects;
        pendingRemoveObjects = [];
        for (var riN = 0; riN < objectsToRemoveNow.length; riN++) {
            try { Main.RemoveObject(objectsToRemoveNow[riN]); }
            catch (e) { logDiagnostic("[DIAG][REMOVE_OBJECT_FAILED] " + e.message); }
        }
        lastRemoveObjectAt = Date.now();
    }
    // [신규][2026-09-09][삭제~다음요청 최소대기 강제] 방금 삭제를 실행했든
    // 이전 사이클에서 실행했든, 마지막 삭제로부터 REMOVE_TO_NEXT_REQUEST_MIN_MS
    // 가 아직 안 지났으면 여기서 멈추고 남은 시간만큼 기다렸다가 이
    // 함수를 다시 부른다(요청은 안 보냄). nEventID=7 전용 타이머 사용.
    if (lastRemoveObjectAt > 0) {
        var sinceRemoveMs = Date.now() - lastRemoveObjectAt;
        if (sinceRemoveMs < REMOVE_TO_NEXT_REQUEST_MIN_MS) {
            Main.SetTimer(7, REMOVE_TO_NEXT_REQUEST_MIN_MS - sinceRemoveMs);
            return;
        }
    }
    if (currentIndex >= targetList.length) {
        // [근본수정][2026-09-25][v5.1][재시도큐 잔존(stale) 엔트리 정리]
        // completedCycles[cycle]이 참이면 push 시점에 건너뛰도록 이미
        // 고쳤지만(재조준 분기), 그건 "이 cycle을 큐에 넣으려는 바로 그
        // 순간" 기준일 뿐이다 - 어떤 cycle X가 이번 패스 앞쪽에서 먼저
        // 실패해 큐에 들어간 "뒤에", 완전히 다른 요청이 재조준되어 X를
        // 채워버리면(completedCycles[X]=true), 이미 큐에 들어가 있던 X의
        // 엔트리는 그대로 남아있었다(등록 당시엔 정당한 실패였으므로) -
        // 그래서 [진행상황] 분자(filledCycles)와 "실패했던 N개 재시도"
        // 숫자가 여전히 앞뒤가 안 맞아 보이는 원인이었다(사용자 재확인).
        // 큐를 실제로 소비하기 직전에 한 번 더 걸러서, 그 사이 다른
        // 경로로 이미 채워진 cycle은 재시도 없이 조용히 제외한다(그
        // cycle은 이미 정상적으로 filledCycles에 반영돼있으므로 데이터
        // 유실 없음).
        retryQueue = retryQueue.filter(function (t) { return !completedCycles[t.cycle]; });
        if (retryQueue.length > 0) {
            logMsg("? [1안] 실패했던 " + retryQueue.length + "개 타임프레임 이번 순환 안에서 재시도합니다...");
            targetList = targetList.concat(retryQueue);
            retryQueue = [];
        } else {
            currentIndex = 0;
            totalPassesCompleted++;
            retryAttemptsThisPass = {}; // [2026-09-02][재시도보완] 순환마다 초기화
            var degradedList = Object.keys(degradedPeriods);
            // [근본수정][2026-09-25][v5.2][순환완료 집계 명시] "저품질 유지중"
            // 로그는 degradedList.length>0일 때만 찍혀서, 화면 스크롤/캡처에서
            // 그 줄을 놓치면 분자(filledCycles)와 총원 사이 나머지가 어디로
            // 갔는지 확인할 길이 없다는 지적(반복) - filledCycles를 초기화
            // 하기 "전에" 최종 집계를 먼저 구해서, 순환완료 로그 자체에
            // "실제수신 X + 저품질 Y = 총Z"를 항상 박아넣는다(저품질 0개여도
            // 표시) - 이 한 줄만 봐도 산수가 맞는지 바로 검증 가능해짐.
            var _finalFilledCount = Object.keys(filledCycles).length;
            var _finalDegradedCount = degradedList.length;
            if (degradedList.length > 0) logMsg("!! [1안] " + degradedList.length + "개 타임프레임 저품질(degraded) 유지중: " + degradedList.join(", "));
            targetList = baseTargetList; // [2026-09-02][재시도보완] 재시도로 늘어났던 targetList 원복
            completedCycles = {};
            filledCycles = {}; // [2026-09-25 신규] 새 순환 시작이니 실제채움 카운터도 같이 초기화
            logMsg("★ [" + totalPassesCompleted + "번째 순환 완료] " + baseTargetList.length + "개 타임프레임 전부 재조회+재계산 끝(실제수신 " + _finalFilledCount + " + 저품질 " + _finalDegradedCount + " = " + (_finalFilledCount + _finalDegradedCount) + ") - 처음부터 다시 순환합니다.");
            // [근본수정][2026-09-03][워밍업 미완료 상태에서 하트비트 조기시작]
            // 예전엔 Main_OnStart가 processNext() 호출 직후 곧바로 하트비트
            // 타이머를 무장해서, 워밍업이 채 끝나기도 전에 하트비트 줄이
            // 나가기 시작했다(워밍업을 안 하고 하트비트로 바로 나오는 것처럼
            // 보이는 원인). 최초 1바퀴(전체 타임프레임) 완료 시점에만 여기서
            // 하트비트를 시작한다.
            if (totalPassesCompleted === 1) {
                Main.SetTimer(4, HEARTBEAT_INTERVAL_MS);
                logMsg("? [워밍업 완료] 이제부터 하트비트를 시작합니다.");
            }
        }
    }
    var target = targetList[currentIndex];
    if (completedCycles[target.cycle]) {
        currentIndex++;
        Main.SetTimer(1, pacingMsCurrent);
        return;
    }
    var fileLabel = formatLabel(target.cycle);
    delete degradedRecoveryPending[fileLabel]; // [2026-09-02][degraded복구 보완]
    var label = INSTRUMENT_TAG + " " + fileLabel;
    currentLabel = label; currentFileLabel = fileLabel; currentCycle = target.cycle; currentItemName = target.nameEn; waitingForResponse = true;
    logDiagnostic("[DIAG][REQUEST_SENT] [" + (currentIndex + 1) + "/" + targetList.length + "] " + label + " 요청 중... (페이싱=" + pacingMsCurrent + "ms)");
    try {
        var reqCount = computeAdaptiveRequestCount(target.cycle, fileLabel);
        // [예스스팟 예시 문법] new ReqChartItem(code, cycle, period, count, countKind)
        var req = new ReqChartItem(target.code, target.cycle, CHART_PERIOD_TICK, reqCount, CHART_REQCOUNT_BAR);
        // [신규][v4.5][프레임 밀림 감지용] "#주기" 커스텀 지표(예스스탁
        // 지표편집기에 사전등록 필요: Plot1(DataCompress,"주기구분");
        // Plot2(BarInterval,"세부주기");)를 같이 요청 - 응답이 실제로
        // 몇 틱짜리인지 GetIndicatorData로 직접 확인 가능. GetCycle/
        // GetPeriod가 이 플랫폼에 없어 그동안 사실상 미검증 상태였던
        // 빈틈을 메운다.
        var indSet = new Array(new IndicatorInfo("#주기"));
        var ok = Main.ReqChartEx(req, null, indSet);
        if (!ok) {
            logDiagnostic("[DIAG][REQUEST_FAILED] label=" + fileLabel + " ReqChartEx가 false 반환");
            recordFailureForRetry(target);
            pacingOnCycleFailure(); waitingForResponse = false; currentIndex++; Main.SetTimer(1, pacingMsCurrent); return;
        }
        requestSeq++; activeRequestSeq = requestSeq;
        requestQueue.push({ seq: requestSeq, cycle: target.cycle, fileLabel: fileLabel, itemName: target.nameEn, count: reqCount, sentAt: Date.now() });
        Main.SetTimer(2, REQUEST_TIMEOUT_MS);
    } catch (e) {
        logDiagnostic("[DIAG][REQUEST_EXCEPTION] label=" + fileLabel + " " + e.message);
        recordFailureForRetry(target);
        pacingOnCycleFailure(); waitingForResponse = false; currentIndex++; Main.SetTimer(1, pacingMsCurrent);
    }
}

function Main_OnRcvChartEx(chartEx) {
    if (requestQueue.length === 0) { pendingRemoveObjects.push(chartEx); return; } // [2026-09-02][누수수정] 대기요청 없이 온 응답도 객체는 삭제큐로
    var matchedIdx = -1;
    for (var qi = 0; qi < requestQueue.length; qi++) if (requestQueue[qi].seq === activeRequestSeq) { matchedIdx = qi; break; }
    if (matchedIdx === -1 || !waitingForResponse) { requestQueue.shift(); pendingRemoveObjects.push(chartEx); return; } // [2026-09-02][누수수정] 지각응답(이미 타임아웃 처리됨)의 객체도 삭제큐로
    // [근본수정][2026-09-10][지각응답(유령응답) 오귀속 - 실제 원인 확인 및 정식수정]
    // chartEx.GetCode(1)/GetCycle(1)/GetPeriod(1)로 이 응답이 실제로 어느
    // 종목코드·주기·구분(분/틱)의 데이터인지 확인해서, 지금 기다리는 요청
    // (reqCtx)의 조건과 다르면 "타임아웃으로 이미 포기했던 요청의 지각응답
    // (유령응답)"으로 보고 폐기하고 계속 기다린다(큐에서 먼저 꺼내지 않음
    // - 유령응답이면 진짜 응답이 아직 큐에 남아있어야 함). 나스닥100

        for (var s = 0; s < signals.length; s++) {
            try {
                var timeStr = signals[s].stime;
                var canonicalMatch = findPendingMatch(thisItemName, thisFileLabel, signals[s].kind, signals[s].sdate, timeStr, signals[s].price);
                var writeSdate = canonicalMatch ? canonicalMatch.sdate : signals[s].sdate;
                var writeStime = canonicalMatch ? canonicalMatch.stime : timeStr;
                var ln = buildSignalLine(thisItemName, thisFileLabel, signals[s].kind, writeSdate, writeStime, signals[s].price, signals[s].leftFlatStar, null, ((signals[s].apexSdate || "") + " " + (signals[s].apexStime || "")).trim(), signals[s].apexHeight, signals[s].leftFlatRun, signals[s].apexSharp, signals[s].isPrettyTriangle);
                if (ln) {
                    var trialSig = buildNewSignalObject(thisItemName, thisFileLabel, signals[s].kind, writeSdate, writeStime, signals[s].price, thisCycle, signals[s].barCountAtSignal, 0, writeSdate + " " + writeStime, null, signals[s].leftFlatStar, signals[s].trueSdate, signals[s].trueStime);
                    var trialResult = resolveHistorically(trialSig, barsForCalc);
                    if (trialResult) {
                        // [2026-08-18 신규][즉시판정 신호 로그 경량화] 감시목록에
                        // 단 한 틱도 안 머물고 등록과 동시에 판정난 신호는, 예전처럼
                        // 신호감지+STATUS_ACHIEVED/INVALID+REMOVAL_LOG 3줄을 남기지
                        // 않는다 - 애초에 대시보드 감시목록에 표시된 적이 없는
                        // 신호라 "제거 기록(REMOVAL_LOG)"도 의미가 없다. 대신
                        // 통계 집계(bridge_signals.py compute_report) 전용으로
                        // 압축된 한 줄(STATS_IMMEDIATE)만 남긴다.
                        // 형식: item,tf,STATS_IMMEDIATE,sdate,stime,price,ACHIEVED|INVALID,소요분,kind
                        if (trialResult.result === "ACHIEVED") achievedCount++; else invalidCount++;
                        var durationMin = null;
                        try { durationMin = (dateTimeToEpochApprox(String(trialResult.sdate), trialResult.stime) - trialSig.signalEpoch) / 60; } catch (eDur) { durationMin = null; }
                        var statsLine = trialSig.itemName + "," + trialSig.tf + ",STATS_IMMEDIATE," + trialSig.sdate + "," + trialSig.stime + "," + trialSig.price.toFixed(2) + "," + trialResult.result + "," + (durationMin !== null ? durationMin.toFixed(2) : "") + "," + trialSig.kind;
                        newSignalLines.push(statsLine);
                        recordResolution(trialSig.tf, trialSig.kind, trialSig.sdate, trialSig.stime, trialSig.price, trialResult.result, trialResult.sdate + " " + trialResult.stime);
                    } else {
                        newSignalLines.push(ln);
                        registerPendingSignal(thisItemName, thisFileLabel, signals[s].kind, writeSdate, writeStime, signals[s].price, thisCycle, signals[s].barCountAtSignal, trialSig.barsElapsedSinceRegistration, "WARM" + frameShiftNote, signals[s].leftFlatStar, signals[s].trueSdate, signals[s].trueStime, trialSig);
                    }
                }
            } catch (e) {}
        }
        writeLinesToFile(newSignalLines);

        // [제거][사용자 요청][철회기능 전체 제거] 여기 있던 철회판정 루프(마진기반 철회/요청봉수부족 감시목록제외 포함)를 통째로 삭제함 - pendingSignals는 이제 이 사이클에서 줄어들지 않음.

        totalCyclesCompleted++;
        if (barsForCalc.length > 0) {
            var lastBar = barsForCalc[barsForCalc.length - 1];
            logDiagnostic("[DIAG][CYCLE_DONE] label=" + thisFileLabel + " 마지막봉 high=" + lastBar.high + " low=" + lastBar.low, lastBar.sdate, lastBar.stime);
            var newestEpochThisCycle = dateTimeToEpochApprox(String(lastBar.sdate), lastBar.stime);
            var prevInfo = lastCycleBarInfo[thisFileLabel];
            if (prevInfo && dateTimeToEpochApprox(String(bars[0].sdate), bars[0].stime) > prevInfo.newestEpoch) {
                logDiagnostic("[DIAG][ADAPTIVE_GAP_DETECTED] label=" + thisFileLabel + " 이전마지막봉=" + prevInfo.newestEpoch + " 이번최초봉epoch=" + dateTimeToEpochApprox(String(bars[0].sdate), bars[0].stime) + " - 다음 요청이 이 공백만큼 자동으로 더 받아옵니다");
            }
            lastCycleBarInfo[thisFileLabel] = { newestEpoch: newestEpochThisCycle };
        }

        // [2026-08-13 안전성수정] 삭제타이머(5)는 이제 Main_OnTimer 안에서
        // 스스로 고정주기로 재무장하므로 여기서 따로 SetTimer를 걸 필요가
        // 없다(걸어도 무해하지만, 요청과 무관하게 독립적으로 도는 게
        // 설계의도라 중복호출을 남기지 않는다).
        pendingRemoveObjects.push(chartEx);
        // [근본수정][2026-09-29][침묵 완료처리 사각지대] 예전엔 예외만 없으면
        // 봉을 0개 받았어도 completedCycles=true로 찍어서 "완료"로 넘어갔다 -
        // 재시도큐에도 안 들어가고 저품질(degraded)로도 안 넘어가는 사각지대라,
        // [진행상황]/[순환완료] 로그의 분자 합이 총원과 안 맞는 걸로 실사례
        // (분봉 M0001~M0007, 104/110)로 확인됨. 봉이 0개면 예외와 똑같이
        // "실패"로 보고 재시도큐로 보낸다 - recordFailureForRetry가 기존
        // 재시도횟수 카운팅을 그대로 하고, MAX_RETRY_ATTEMPTS를 다 채우면
        // 그 함수 스스로 degradedPeriods+completedCycles를 true로 만든다
        // (기존 저품질 전환 로직 재사용, 새 분기 없음).
        if (bars.length > 0) {
            completedCycles[thisCycle] = true;
            filledCycles[thisCycle] = true; // [2026-09-25 신규] 예외 없이 여기까지 왔고 봉을 1개 이상 실제로 받았을 때만 true - thisCycle은 #주기 프레임재조준이 있었으면 이미 실제값으로 치환된 상태라 그대로 정확함
        } else {
            recordFailureForRetry({ code: MARKET_DATA_CODE, nameEn: thisItemName, cycle: thisCycle });
        }

    } catch (e) {
        logMsg("  !! 처리오류: " + thisFileLabel + " - " + e.message);
        recordFailureForRetry({ code: MARKET_DATA_CODE, nameEn: thisItemName, cycle: thisCycle });
        pendingRemoveObjects.push(chartEx);
    }

    if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
    pacingOnCycleSuccess();
    currentIndex++;
    // [임시 진단][2026-09-05][하트비트 시작 전엔 화면이 조용해서 멈춘
    // 것처럼 보이는 문제] 워밍업 1바퀴가 다 끝나야 하트비트가 시작되는데,
    // 그 전까지는 개별 요청 로그가 전부 화면에 안 뜨는 조용한 태그라서,
    // 실제로는 정상 진행 중이어도 화면만 보면 멈춘 것처럼 보일 수
    // 있었다(실제 지적사항). 10개 완료마다 눈에 보이는 진행상황을
    // 남겨서, 조용히 죽은 건지 정상 진행 중인지 바로 구분 가능하게 한다.
    // [2026-09-25 수정][실제채움 기준으로 정정] completedCycles(성공+degraded
    // 포기 합산)를 그대로 "실제 채워진 개수"라고 표시하던 게 거짓정보였음
    // (검증 완료). filledCycles(진짜 데이터 받은 것만)를 분자로 쓰고,
    // degraded 포기분은 별도로 명시한다.
    if (currentIndex % 10 === 0 || currentIndex === targetList.length) {
        var _degradedNowCount = Object.keys(degradedPeriods).length;
        logMsg("[진행상황] " + Object.keys(filledCycles).length + "/" + baseTargetList.length + " 타임프레임 실제 데이터 수신 완료" +
            (_degradedNowCount > 0 ? " (그 외 저품질(degraded)로 포기 " + _degradedNowCount + "개 - 완료 처리는 됐으나 실데이터 아님)" : ""));
    }
    Main.SetTimer(1, pacingMsCurrent);
}

function Main_OnRcvMarketData(MarketData) { marketDataObj = MarketData; }
function Main_OnUpdateMarket(itemcode, updateID, exchangeKind) {
    if (!marketDataObj || itemcode !== MARKET_DATA_CODE) return;
    if (updateID !== 20001) return;
    totalTicksReceived++;
    var curPrice = Number(marketDataObj.current);
    if (!isFinite(curPrice) || curPrice <= 0) return;
    var sdate = parseDateField(marketDataObj.date), stime = parseTimeField(marketDataObj.time);
    if (!sdate || !stime) return;
    lastTickEpoch = dateTimeToEpochApprox(sdate, stime);
    if (REALTIME_TRACK_ENABLED) {
        Object.keys(liveBuilders).forEach(function (periodStr) {
            try {
                var period = Number(periodStr);
                var bar = liveBuilders[period].onTick(curPrice, sdate, stime);
                if (bar) processLiveBar(period, bar.high, bar.low, bar.sdate, bar.stime);
            } catch (e) {}
        });
        Object.keys(realtimeOnlyBuilders).forEach(function (periodStr) {
            try {
                var period = Number(periodStr);
                var bar = realtimeOnlyBuilders[period].onTick(curPrice, sdate, stime);
                if (bar) processRealtimeOnlyBar(period, bar.high, bar.low, bar.sdate, bar.stime);
            } catch (e) {}
        });
        try { updatePendingSignals(curPrice, sdate, stime); } catch (e) {}
    }
}

    if (nEventID == 1) { Main.KillTimer(1); processNext(); }
    else if (nEventID == 2) {
        Main.KillTimer(2);
        if (waitingForResponse) {
            waitingForResponse = false;
            var timedOutReq = null;
            for (var qi2 = requestQueue.length - 1; qi2 >= 0; qi2--) if (requestQueue[qi2].seq === activeRequestSeq) { timedOutReq = requestQueue.splice(qi2, 1)[0]; break; }
            timeoutTrackedLabels[currentFileLabel] = true; // [신규][2026-09-10] 이 라벨의 다음 정상매칭까지 추적
            pacingOnCycleFailure();
            currentIndex++;
            Main.SetTimer(1, pacingMsCurrent + TIMEOUT_COOLDOWN_MS);
        }
    } else if (nEventID == 3) {
        Main.KillTimer(3); quickSaveScheduled = false;
        if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
    } else if (nEventID == 4) {
        Main.KillTimer(4);
        var elapsedMin = scriptStartTime ? Math.floor((Date.now() - scriptStartTime) / 60000) : -1;
        logMsg("[하트비트] 경과:" + elapsedMin + "분 / 대기중:" + pendingSignals.length + " / 달성:" + achievedCount + " / 무효:" + invalidCount +

            " recentResolutions=" + Object.keys(recentResolutions).length + " orphanedSignalDetails=" + Object.keys(orphanedSignalDetails).length +
            " lastSeenRealTime=" + Object.keys(lastSeenRealTime).length);
        pruneDedupStructures();
        flushDiagnosticLog();
        try {
            var hbDate = parseDateField(marketDataObj ? marketDataObj.date : 0);
            var hbTime = parseTimeField(marketDataObj ? marketDataObj.time : null) || "00:00:00";
            timedPrintOnFile(OUTPUT_FILE, (currentItemName || INSTRUMENT_NAME) + ",HEARTBEAT_TICK,HEARTBEAT," + hbDate + "," + hbTime + ",0," + buildPendingKeysString() + "," + buildPendingDetailsString() + "\n");
        } catch (e) {}
        if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
        Main.SetTimer(4, HEARTBEAT_INTERVAL_MS);
    } else if (nEventID == 5) {
        // [2026-08-13 안전성수정] 기존엔 이 핸들러가 스스로를 재무장하지
        // 않고, Main_OnRcvChartEx 성공/실패 분기마다 외부에서
        // Main.SetTimer(5, REMOVE_OBJECT_DELAY_MS)를 걸어 재무장하는
        // 구조였다. 1안은 targetList를 끝까지 돌면 처음부터 다시
        // 순환하는 영구루프이고, pacingMsCurrent가 결국 0ms까지 떨어져
        // 그대로 유지되므로(PACING_MS_MIN=0), 정상운영 상태에서는 다음
        // 사이클이 300ms(REMOVE_OBJECT_DELAY_MS)보다 먼저 도착해 삭제
        // 타이머가 발동 직전에 계속 리셋될 수 있었다(같은 타이머ID를
        // 반복 호출하는 건 이 코드 전체에서 "재무장"으로 쓰이므로) -
        // 이러면 Main.RemoveObject가 사실상 영원히 호출 안 되고
        // pendingRemoveObjects가 무한정 쌓여 차트객체 누적으로 다운될
        // 여지가 있었다. 이제는 이 핸들러가 매번 자기 자신을 다시
        // 무장해서(요청 사이클과 무관하게) 고정주기로 독립적으로
        // 발동하므로, 요청이 아무리 빨리 몰려도 삭제가 굶주리지 않는다.
        if (pendingRemoveObjects.length > 0) {
            var objectsToRemove = pendingRemoveObjects;
            pendingRemoveObjects = [];
            for (var ri5 = 0; ri5 < objectsToRemove.length; ri5++) {
                try { Main.RemoveObject(objectsToRemove[ri5]); }
                catch (e) { logDiagnostic("[DIAG][REMOVE_OBJECT_FAILED] " + e.message); }
            }
            lastRemoveObjectAt = Date.now();
        }
        Main.SetTimer(5, REMOVE_OBJECT_DELAY_MS);
    } else if (nEventID == 7) {
        // [신규][2026-09-09][삭제~다음요청 최소대기용 재시도 타이머]
        Main.KillTimer(7);
        processNext();
    } else if (nEventID == 6) {
        // [2026-09-02][degraded복구 보완] 분봉본과 동일 패턴
        if (!waitingForResponse) {
            var degradedList6 = Object.keys(degradedPeriods);
            for (var dgi = 0; dgi < degradedList6.length; dgi++) {
                var dLabel = degradedList6[dgi];
                if (degradedRecoveryPending[dLabel]) continue;
                var dCycle = parseInt(dLabel.slice(1), 10);
                degradedRecoveryPending[dLabel] = true;
                var insertAt = Math.min(currentIndex, targetList.length);
                targetList.splice(insertAt, 0, { code: MARKET_DATA_CODE, nameEn: INSTRUMENT_NAME, cycle: dCycle });
                delete completedCycles[dCycle]; // [신규][사용자 지적으로 발견한 충돌 수정] 포기시 completedCycles에 "임시완료"로 표시해뒀던 걸 진짜 재시도 직전에 해제 - 안 그러면 스킵로직이 이 재삽입을 건너뜀
                logMsg("  ? [degraded복구] " + dLabel + " 본 순환에 재삽입(1분 주기)");
                break;
            }
        }
        Main.SetTimer(6, DEGRADED_RECOVERY_INTERVAL_MS);
    }
