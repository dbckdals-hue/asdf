function recordRealBarInterval(label, barEpochNow) {

    if (prevEpoch !== undefined && prevEpoch !== null && barEpochNow > prevEpoch) {
        if (!recentBarIntervals[label]) recentBarIntervals[label] = [];
        recentBarIntervals[label].push(barEpochNow - prevEpoch);
        if (recentBarIntervals[label].length > RECENT_BAR_INTERVAL_WINDOW) recentBarIntervals[label].shift();
    }
    lastRealBarEpochByLabel[label] = barEpochNow;
}
function recentAvgBarIntervalSec(label) {

    var sum = 0;
    for (var i = 0; i < arr.length; i++) sum += arr[i];
    return sum / arr.length;
}

    // 통일] 1분봉만 순수 300 고정(오래된신호 여유 무시)으로 뒀던 건
    // 테스트 목적의 임시 특례였다 - 정식으로 구간을 다시 나눔:
    // 5분미만=800, 10분미만=400, 30분미만=300, 50분미만=150, 50분이상=100.
    // 전 구간 오래된신호 여유(+60)도 동일하게 적용받는다.
    // [수정][2026-09-09][활발한 시간대 여유치 1.5배] 기존
    // isPeakVolumeWindowKST()(21~24시, 원래는 객체생존확인 주기 조절용)를
    // 요청봉수 계산에도 재사용 - 틱봉 v1.34와 동일 원칙.
    var baseFloor = cycle < 5 ? 800 : (cycle < 10 ? 400 : (cycle < 30 ? 300 : (cycle < 50 ? 150 : 100)));
    if (isPeakVolumeWindowKST()) baseFloor = Math.ceil(baseFloor * 1.5);
    if (info) {
        var nowEpoch = lastTickEpoch || Math.floor(Date.now() / 1000);
        var elapsedSec = Math.max(1, nowEpoch - info.newestEpoch);
        var expectedNewBars = Math.ceil(elapsedSec / (cycle * 60));
        adaptiveCount = Math.ceil(expectedNewBars * 1.5) + baseFloor;
    }
    var coverageFloor = baseFloor;
    var oldestEpoch = oldestPendingSignalEpoch(fileLabel);
    if (oldestEpoch !== null) {
        var nowEp = lastTickEpoch || Math.floor(Date.now() / 1000);
        var barsSinceOldest = Math.ceil(Math.max(0, nowEp - oldestEpoch) / (cycle * 60));
        coverageFloor = Math.max(coverageFloor, barsSinceOldest + 60); // 여유 60봉
    }
    if (windowNeedsFull[fileLabel]) return Math.min(MAX_BAR_COUNT, Math.max(originalMax, coverageFloor)); // [v5.17] 직전 응답이 가장 오래된 감시신호를 못 덮음 -> 최대 크기
    if (adaptiveCount === null) return Math.min(MAX_BAR_COUNT, Math.max(originalMax, coverageFloor));
    return Math.min(MAX_BAR_COUNT, Math.max(coverageFloor, Math.min(originalMax, adaptiveCount)));
}

function pruneDedupStructures() {
    if (lastTickEpoch === null) return;
    var cutoff = lastTickEpoch - DEDUP_RETENTION_SEC;
    Object.keys(pretty2Written).forEach(function (k) { if (pretty2Written[k].epoch < cutoff) delete pretty2Written[k]; });
    var _pkSet = {}; pendingSignals.forEach(function (s) { _pkSet[pkOf(s)] = true; }); Object.keys(gateSyncBase).forEach(function (k) { if (!_pkSet[k]) delete gateSyncBase[k]; });
    var removedSeenKeys = 0, removedPriceEntries = 0, removedPriceKeys = 0, removedResolutions = 0, removedOrphaned = 0;
    Object.keys(seenKeys).forEach(function (k) {
        if (seenKeys[k] < cutoff) { delete seenKeys[k]; removedSeenKeys++; }
    });
    Object.keys(seenKeysByPrice).forEach(function (pk) {
        var list = seenKeysByPrice[pk];
        var kept = list.filter(function (entry) { return entry.epoch >= cutoff; });
        removedPriceEntries += (list.length - kept.length);
        if (kept.length === 0) { delete seenKeysByPrice[pk]; removedPriceKeys++; }
        else seenKeysByPrice[pk] = kept;
    });
    Object.keys(recentResolutions).forEach(function (k) {
        if (recentResolutions[k].epoch < cutoff) { delete recentResolutions[k]; removedResolutions++; }
    });
    Object.keys(orphanedSignalDetails).forEach(function (k) {
        if (orphanedSignalDetails[k].epoch < cutoff) { delete orphanedSignalDetails[k]; removedOrphaned++; }
    });
    if (removedSeenKeys || removedPriceEntries || removedPriceKeys || removedResolutions || removedOrphaned) {
        logDiagnostic("[DIAG][DEDUP_PRUNE] seenKeys=-" + removedSeenKeys + " priceEntries=-" + removedPriceEntries + " priceKeys=-" + removedPriceKeys + " resolutions=-" + removedResolutions + " orphaned=-" + removedOrphaned +
            " (잔존: seenKeys=" + Object.keys(seenKeys).length + " priceKeys=" + Object.keys(seenKeysByPrice).length + " resolutions=" + Object.keys(recentResolutions).length + " orphaned=" + Object.keys(orphanedSignalDetails).length + ")");
        pendingStateDirty = true;
    }
}

function SlidingExtreme(size, mode) { this.size = size; this.mode = mode; this.buf = []; this.idx = 0; }
SlidingExtreme.prototype.push = function (val) {
    var i = this.idx++;
    var better = (this.mode === 'max') ? function (a, b) { return a <= b; } : function (a, b) { return a >= b; };
    while (this.buf.length && better(this.buf[this.buf.length - 1].v, val)) this.buf.pop();
    this.buf.push({ v: val, i: i });
    while (this.buf.length && this.buf[0].i <= i - this.size) this.buf.shift();
    return this.buf[0].v;
};

    // 위에서 이미 별표(star) 계산에 쓰던 "SB가 몇 봉째 안 바뀌었는지" 원시
    // 봉수값이다 - 대시보드가 별표(0/0.5/1 3단계)보다 더 구체적인 실제
    // 봉수를 "선행평탄구간"으로 보여줄 수 있게 그대로 반환값에 실어보낸다.
    // b1/s1이 모두 false면 flatRun1은 계산 자체가 안 돌아 undefined인 채로
    // 반환되지만, 호출부(onBar)는 st.b1/st.s1이 true일 때만 이 값을 쓴다.
    return { b1: b1, b2: b2, s1: s1, s2: s2, b1Star: b1Star, b2Star: b2Star, s1Star: s1Star, s2Star: s2Star, lf1: lf1, lf2: lf2, flatRun1: flatRun1, flatRun2: flatRun2 };
}
// [신규][2026-09-10][목표삼각형 뾰족여부 판정기준] 폭2(BULL2/BEAR2) 신호는
// 후보 꼭지점이 index1/index2 둘이라, 더 높은 쪽을 "진짜 꼭지점"으로 쓴다
// (기존 로직 그대로). 그런데 두 후보 높이가 서로 비슷하면 실제로는 뾰족한
// 한 지점이 아니라 두 봉짜리 뭉툭한 고원(plateau)에 가깝다 - 두 후보 높이
// 차이가 (더 높은 쪽 대비) 이 비율 이상 벌어져야 "뾰족"으로 표시한다.
// 폭1(BULL1/BEAR1)은 후보가 하나뿐이라 항상 뾰족으로 취급.
var APEX_SHARP_HEIGHT_RATIO = 0.15;
// [신규][v4.4][예쁜삼각형(다이아몬드 배지) 판정 - GOLD 최신본에서 이식]
// 기존 Bear_1/2/3·Bull_1/2/3(계곡패턴) 조건은 이미 evalSignal이 판정한다.
// 여기서는 그 위에 "더 확실한 모양"만 추가로 가려내는 보조검사다:
// (1)골(계곡)이 충분히 뾰족한지(펑퍼짐 걸러냄) (2)그 직전에 이미 반대편
// 언덕/골짜기가 있었는지(빈약한 이전구름 걸러냄) (3)회복 직후 구름이
// 계속 벌어지는 방향인지. saHist/sbHist는 index0=현재봉, index1=1봉전...
// 순서라(unshift로 쌓임) 사용자가 준 수식의 SA(N)/SB(N) 표기와 그대로
// 대응된다. isBull=true면 Bull(선행스팬A가 평탄, SB가 위에서 찌르고
// 회복), false면 Bear(선행스팬B가 평탄, SA가 아래서 찌르고 회복).
function computeAtrFromHist(saHist) {
    var atr = 0;
    for (var i = 0; i < 14 && i < saHist.length - 1; i++) atr += Math.abs(saHist[i] - saHist[i + 1]);
    return Math.max(atr / 14, 0.0001);
}
function checkPrettyTriangle(saHist, sbHist, isBull) {
    if (saHist.length < 8 || sbHist.length < 8) return false;
    var atr = computeAtrFromHist(saHist);
    var threshold = atr * PRETTY_TRIANGLE_RATIO;
    if (!isBull) {
        // Bear: 골 깊이(SB - 최근3봉 중 SA 최저치), 이전언덕(SA가 SB보다 위였던 5~7봉전)
        var golDepth = sbHist[0] - Math.min(saHist[0], saHist[1], saHist[2]);
        var prevHill = (saHist[5] > sbHist[5]) || (saHist[6] > sbHist[6]) || (saHist[7] > sbHist[7]);
        var expanding = (saHist[0] - sbHist[0]) > (saHist[1] - sbHist[1]);
        return golDepth >= threshold && prevHill && expanding;
    } else {
        // Bull: 골 깊이(최근3봉 중 SB 최고치 - SA), 이전골짜기(SB가 SA보다 아래였던 5~7봉전)
        var golDepthBull = Math.max(sbHist[0], sbHist[1], sbHist[2]) - saHist[0];
        var prevValley = (sbHist[5] < saHist[5]) || (sbHist[6] < saHist[6]) || (sbHist[7] < saHist[7]);
        var expandingBull = (saHist[0] - sbHist[0]) > (saHist[1] - sbHist[1]);
        return golDepthBull >= threshold && prevValley && expandingBull;
    }
}
var PRETTY_TRIANGLE_RATIO = 0.15; // 실측하며 조절할 값(높을수록 더 뾰족한 것만 통과)
// [신규][2026-09-26][흰다이아몬드 배지 - 오른쪽 구름대 확인] 기존
// 다이아몬드(checkPrettyTriangle)는 그대로 두고, "삼각형 완성 후 반대편에도
// 구름대가 형성되는지" 추가 배지 - 새 스캔/새 차트요청 없이 이미 매 봉마다
// 호출되는 이 SignalCalculator.onBar()의 saHist/sbHist 누적을 재사용해서
// "꼭짓점 이후 N봉이 쌓일 때까지 기다렸다가 그 구간 구름대 두께가
// 유지되는지"만 사후 확인한다(신규 신호검색 기준 아님).
var PRETTY_TRIANGLE2_LOOKAHEAD_BARS = 8; // checkPrettyTriangle의 왼쪽 8봉 대칭
// [수정][2026-09-26][v5.8 - 오른쪽 구름대 판정을 독립된 케이스들로 분리]
// 사용자 지적: 얇아도 평탄구간이 넓고 삼각형이 예쁘면(=기존 다이아몬드를
// 이미 통과) 상관없는데 v5.7의 "평균 threshold 미달"이 이런 경우까지
// 탈락시킴 - 그리고 실제 차트엔 넓은 평탄구간 안에서 잠깐 한 번
// 아래로/위로 삐끗(V자)했다가 회복하는 형태도 있음. 사용자 결정: 여러
// 기준을 하나로 합치지 말고 "케이스"로 서로 완전히 독립시켜서, 나중에
// 특정 케이스만 골라 끄거나 되돌릴 수 있게 하고, 통과한 케이스 번호를
// 그대로 기록해 배지 마우스오버에 "몇 번 케이스로 통과했는지" 보여준다
// (중복 통과시 "1,2"처럼 여러 개). 각 케이스는 서로를 몰라도 되는 순수
// 함수 - 하나를 고치거나 빼도 나머지에 영향 없음.
function checkRightCloudCase1_ThickAverage(saHist, sbHist, isBull) {
    // 케이스1(두께평균 - 가장 엄격/확실): 한 봉이라도 두께가 0 이하(구름
    // 소멸/역전)면 무조건 탈락(하드 플로어), 나머지는 N봉 평균 두께가
    // threshold 이상이면 통과 - v5.7 로직 그대로, 케이스1로 이름만 분리.
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    var atr = computeAtrFromHist(saHist);
    var threshold = atr * PRETTY_TRIANGLE_RATIO;
    var sumThickness = 0;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) return false;
        sumThickness += thickness;
    }
    return (sumThickness / PRETTY_TRIANGLE2_LOOKAHEAD_BARS) >= threshold;
}
function checkRightCloudCase2_ThinButPresent(saHist, sbHist, isBull) {
    // 케이스2(얇아도 유지): 두께 크기(threshold)는 아예 안 보고, N봉 내내
    // 단 한 번도 역전/소멸하지 않았는지만 확인 - 케이스1보다 완화된 조건
    // (케이스1을 통과하면 이 케이스도 항상 자동으로 통과하는 상위개념).
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) return false;
    }
    return true;
}
var PRETTY_TRIANGLE2_DIP_TOLERANCE_BARS = 2; // N=8봉 중 최대 허용 역전/소멸 봉수(실측하며 조절할 값)
function checkRightCloudCase3_BriefDipTolerant(saHist, sbHist, isBull) {
    // 케이스3(짧은 V자 역전 허용): N봉 중 최대 DIP_TOLERANCE개 봉까지는
    // 역전/소멸돼도 되고, 나머지(절대다수)만 정상 방향이면 통과 - 넓은
    // 평탄구간 안에서 잠깐 한 번 삐끗했다 회복하는 형태까지 인식(케이스2를
    // 통과하면 이 케이스도 항상 자동으로 통과하는 상위개념).
    if (saHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS || sbHist.length < PRETTY_TRIANGLE2_LOOKAHEAD_BARS) return false;
    var invertedCount = 0;
    for (var i = 0; i < PRETTY_TRIANGLE2_LOOKAHEAD_BARS; i++) {
        var thickness = isBull ? (sbHist[i] - saHist[i]) : (saHist[i] - sbHist[i]);
        if (thickness <= 0) invertedCount++;
    }
    return invertedCount <= PRETTY_TRIANGLE2_DIP_TOLERANCE_BARS;
}
// 세 케이스를 서로 독립적으로 평가해 통과한 케이스 번호만 모아 반환 -
// 어느 하나라도 통과하면 흰다이아몬드 확정(matched=true). cases 배열은
// 그대로 신호파일/대시보드까지 실어보내 배지 마우스오버에 표시한다.
function evalRightCloudCases(saHist, sbHist, isBull) {
    var cases = [];
    if (checkRightCloudCase1_ThickAverage(saHist, sbHist, isBull)) cases.push(1);
    if (checkRightCloudCase2_ThinButPresent(saHist, sbHist, isBull)) cases.push(2);
    if (checkRightCloudCase3_BriefDipTolerant(saHist, sbHist, isBull)) cases.push(3);
    return { matched: cases.length > 0, cases: cases };
}

// [신규][2026-10-01][v5.21 다이아몬드(보라/흰색) 기준 개편] 사용자 설계:
//  - 신호 구조: 평탄한 SB 위에서 SA가 삼각형으로 솟는 모양이고, 삼각형 양옆에는 삼각형과 "반대색" 구름(BULL이면 SA<SB, BEAR면 SA>SB)이 있다.
//    (삼각형 색이 구름과 반대인 것은 신호 자체이므로 제외 사유가 아님. 구름이 삼각형 앞뒤로 바로 뒤집히는 유형은 보류)
//  - 보라 💎 = 즉시성: 삼각형 완성 봉에서 바로. 평탄 2봉 이상 + 뾰족한 삼각형 + 왼쪽 8봉 중 4봉 이상에 구름(두께 기준 없음).
//  - 흰색 ◇ = 완성도(보라보다 높은 기준): 보라 통과 신호만 대상. 삼각형 뒤 8봉이 쌓인 시점에
//    왼쪽 8봉 중 6봉 이상 구름 + 평균두께>=기준(신호 시점에 계산), 오른쪽 8봉 중 6봉 이상 구름 + 평균두께>=기준.
//    두께 기준 = ATR(선행스팬A 14봉 평균변화) x PRETTY_TRIANGLE_RATIO(0.15).
//  - 기존 기준(v5.20 이전: checkPrettyTriangle 골깊이/이전언덕/확장 + 흰색 케이스1/2/3)은 함수째 그대로 남겨뒀다.
//    아래 DIAMOND_CRITERIA_LEGACY 를 true 로 바꾸면 기존 기준으로 복원된다.
// ================================================================================
var DIAMOND_CRITERIA_LEGACY = false;   // true = 기존(v5.20 이전) 보라/흰색 기준으로 복원
var DIAMOND_MIN_FLAT_RUN = 2;          // 보라: 평탄구간 최소 봉수
var DIAMOND_PURPLE_LEFT_MIN_BARS = 4;  // 보라: 왼쪽 8봉 중 구름이 있어야 하는 최소 봉수
var DIAMOND_WHITE_SIDE_MIN_BARS = 6;   // 흰색: 왼쪽/오른쪽 8봉 중 구름이 있어야 하는 최소 봉수
function diamondCloudThickness(sa, sb, i, isBull) { return isBull ? (sb[i] - sa[i]) : (sa[i] - sb[i]); } // 삼각형과 반대색 구름이면 양수
function diamondSideStats(sa, sb, from, isBull) {
    var n = PRETTY_TRIANGLE2_LOOKAHEAD_BARS, cnt = 0, sum = 0;
    if (sa.length < from + n || sb.length < from + n) return { cnt: 0, avg: 0, enough: false };
    for (var i = from; i < from + n; i++) {
        var t = diamondCloudThickness(sa, sb, i, isBull);
        if (t > 0) { cnt++; sum += t; }
    }
    return { cnt: cnt, avg: sum / n, enough: true };
}
// 보라: 삼각형 완성 시점. leftFrom = 삼각형 왼쪽 첫 봉 index(폭1 신호=2, 폭2 신호=3)
function checkDiamondPurpleNew(sa, sb, isBull, leftFrom, flatRun, apexSharp) {
    var out = { ok: false, leftStrict: false };
    if (!apexSharp || !(flatRun >= DIAMOND_MIN_FLAT_RUN)) return out;
    var left = diamondSideStats(sa, sb, leftFrom, isBull);
    if (!left.enough || left.cnt < DIAMOND_PURPLE_LEFT_MIN_BARS) return out;
    out.ok = true;
    var threshold = computeAtrFromHist(sa) * PRETTY_TRIANGLE_RATIO;
    out.leftStrict = left.cnt >= DIAMOND_WHITE_SIDE_MIN_BARS && left.avg >= threshold; // 흰색의 왼쪽 조건(신호 시점에 미리 계산해 둠)
    return out;
}
// 흰색: 삼각형 뒤 8봉이 쌓인 시점(index 0..7 = 새로 쌓인 8봉). leftStrict 는 보라 통과 시점에 계산한 값.
function evalDiamondWhiteNew(sa, sb, isBull, leftStrict) {
    if (!leftStrict) return { matched: false, cases: [] };
    var right = diamondSideStats(sa, sb, 0, isBull);
    var threshold = computeAtrFromHist(sa) * PRETTY_TRIANGLE_RATIO;
    var ok = right.enough && right.cnt >= DIAMOND_WHITE_SIDE_MIN_BARS && right.avg >= threshold;
    return { matched: ok, cases: ok ? [4] : [] }; // 케이스 4 = 신기준(왼쪽+오른쪽 구름 확인)
}
function SignalCalculator() {

SignalCalculator.prototype.drainPretty2Events = function () {

SignalCalculator.prototype.onBar = function (high, low, sdate, stime) {
    var h9 = this.ex9h.push(high), l9 = this.ex9l.push(low);
    var h26 = this.ex26h.push(high), l26 = this.ex26l.push(low);
    var h52 = this.ex52h.push(high), l52 = this.ex52l.push(low);
    var tanki = (h9 + l9) / 2, jyunki = (h26 + l26) / 2;
    var SA = (tanki + jyunki) / 2, SB = (h52 + l52) / 2;
    // [신규][2026-09-14][진입 다음봉 재확인용] 이번 봉의 SA/SB를 인스턴스에 노출 -
    // processLiveBar가 "직전 봉에서 막 등록된 신호가 이번 봉에도 삼각형(재관통)을
    // 유지하는지" 확인할 때 이 값을 그대로 읽는다. 검색조건(b1/b2/s1/s2) 자체는
    // 전혀 안 건드림 - 순수 노출용 추가.
    this.lastSA = SA; this.lastSB = SB;
    this.saHist.unshift(SA); if (this.saHist.length > 30) this.saHist.pop();
    this.sbHist.unshift(SB); if (this.sbHist.length > 30) this.sbHist.pop();
    this.sdateHist.unshift(sdate); if (this.sdateHist.length > 30) this.sdateHist.pop();
    this.stimeHist.unshift(stime); if (this.stimeHist.length > 30) this.stimeHist.pop();
    this.barCount++;
    // [신규][2026-09-26][흰다이아몬드] 대기 중인 꼭짓점들을 한 봉씩 나이먹인다 -
    // 이 saHist/sbHist는 신호검색과 완전히 동일한 배열이라 새 조회/새 순회가
    // 전혀 필요 없다. N봉째 되는 바로 이 순간의 index 0..N-1이 정확히
    // "꼭짓점 다음에 새로 쌓인 N개 봉"이다.
    if (this.pendingPretty2.length > 0) {
        for (var p2i = this.pendingPretty2.length - 1; p2i >= 0; p2i--) {
            var pc = this.pendingPretty2[p2i];
            pc.bars++;
            if (pc.bars >= PRETTY_TRIANGLE2_LOOKAHEAD_BARS) {
                var rc = DIAMOND_CRITERIA_LEGACY ? evalRightCloudCases(this.saHist, this.sbHist, pc.isBull) : evalDiamondWhiteNew(this.saHist, this.sbHist, pc.isBull, pc.leftStrict); // [v5.21] 신기준(기존은 LEGACY 플래그로 복원)
                this.pretty2Events.push({ sdate: pc.sdate, stime: pc.stime, price: pc.price, kind: pc.kind, confirmed: rc.matched, cases: rc.cases });
                this.pendingPretty2.splice(p2i, 1);
            }
        }
    }
    var results = [];
    if (this.barCount <= 51) return results;

function dateTimeToEpochApprox(sdate, stime) {

}

function MinuteBarBuilder(nMinutes) { this.n = nMinutes; this.bucket = null; this.high = null; this.low = null; this.lastSdate = null; this.lastStime = null; }
MinuteBarBuilder.prototype.onTick = function (price, sdate, stime, epochSeconds) {
    var bucket = Math.floor(epochSeconds / 60 / this.n);
    var completed = null;
    if (this.bucket === null) { this.bucket = bucket; this.high = price; this.low = price; this.lastSdate = sdate; this.lastStime = stime; return null; }
    if (bucket !== this.bucket) {
        // [신규][2026-09-26][v5.6] 이 틱이 도착하기 전 몇 개의 캘린더 버킷이
        // 통째로 무틱(공백)이었는지 - 정상 진행(바로 다음 버킷)이면 0.
        var gapBars = bucket - this.bucket - 1;
        completed = { high: this.high, low: this.low, sdate: this.lastSdate, stime: this.lastStime, gapBars: gapBars };
        this.bucket = bucket; this.high = price; this.low = price;
    }
    else { if (price > this.high) this.high = price; if (price < this.low) this.low = price; }
    this.lastSdate = sdate; this.lastStime = stime;
    return completed;
};

function countRealBarsAfterEpoch(bars, epochRef) {
    if (epochRef === undefined || epochRef === null) return 0;
    var n = 0;
    for (var i = 0; i < bars.length; i++) {
        if (bars[i].filled) continue;
        if (dateTimeToEpochApprox(String(bars[i].sdate), bars[i].stime) > epochRef) n++;
    }
    return n;
}
function getLiveReseedStats(period) { return liveReseedStats[period] || (liveReseedStats[period] = { n: 0, discarded: 0, gap: 0 }); }
function logGateCountSync(sig, base, preCount, chartCnt) {
    var diff = chartCnt - preCount;
    if (Math.abs(diff) < GATE_SYNC_MIN_DIFF) return;
    var nowEp = lastTickEpoch || Math.floor(Date.now() / 1000);
    if (sig._syncLogEpoch !== undefined && sig._syncLogEpoch !== null && (nowEp - sig._syncLogEpoch) < GATE_SYNC_LOG_COOLDOWN_SEC) return;
    sig._syncLogEpoch = nowEp;
    var isTick = (sig.tf.charAt(0) === "T");
    var rs = getLiveReseedStats(sig.period);
    var liveD = (sig._liveBars || 0) - base.liveBars;          // 직전 동기화 이후 실시간이 완결시킨 봉 수
    var chartD = chartCnt - base.chartCnt;                      // 같은 구간에 차트에 생긴 봉 수
    var reseeds = rs.n - base.reseed, discarded = rs.discarded - base.discard, gaps = rs.gap - base.gap;
    var ticksD = isTick ? (totalTicksReceived - base.ticks) : 0;
    var expTicks = isTick ? chartD * sig.period : 0;
    var cause;
    if (diff > 0) {
        if (isTick && chartD >= 3 && ticksD < 0.8 * expTicks) cause = "TICK_SHORT";                         // 실시간 수신틱이 차트 기준보다 적음
        else if (!isTick && gaps > 0 && chartD - liveD >= 1) cause = "LIVE_GAP_STOP";                       // 무틱공백으로 실시간 정지
        else if (chartD - liveD >= 2 && reseeds > 0 && isTick && discarded >= sig.period * 0.5 * (chartD - liveD)) cause = "RESEED_LOSS"; // 재조회 때 실시간 빌더 재시작으로 손실
        else if (chartD - liveD >= 2) cause = "LIVE_BAR_MISSED";                                            // 실시간이 봉을 놓침(원인 미상)
        else cause = "COUNTER_LOW_OTHER";                                                                   // 구간 봉수는 맞는데 카운터가 작음(기타)
    } else {
        cause = "LIVE_OVERCOUNT";                                                                            // 실시간이 차트보다 봉을 더 셈
    }
    var detail = "sig=" + sig.sdate + " " + sig.stime + "@" + Number(sig.price).toFixed(2) +
        "|liveD=" + liveD + "|chartD=" + chartD + "|pre=" + preCount + "|reseed=" + reseeds +
        (isTick ? ("|discard=" + discarded + "|ticks=" + ticksD + "|expTicks=" + expTicks) : ("|gap=" + gaps));
    try {
        logDiagnostic("[DIAG][GATE_COUNT_SYNC] label=" + sig.tf + " kind=" + sig.kind + " live=" + preCount + " chart=" + chartCnt + " cause=" + cause + " " + detail);
        var wp = nowServerTimeStr().split(" ");
        timedPrintOnFile(OUTPUT_FILE, sig.itemName + "," + sig.tf + ",GATE_COUNT_SYNC," + wp[0] + "," + wp[1] + "," + sig.kind + "," + preCount + "," + chartCnt + "," + cause + "," + detail + "\n");
    } catch (eLog) {}
}
function syncGateCounter(sig, completedBars, preWalkCount) {
    var chartCnt = countRealBarsAfterEpoch(completedBars, sig.signalEpoch);
    var pk = pkOf(sig), rs = getLiveReseedStats(sig.period);

            logDiagnostic("[DIAG][GATE_COUNT_SYNC_INIT] label=" + sig.tf + " kind=" + sig.kind + " saved=" + preWalkCount + " chart=" + chartCnt + " diff=" + (chartCnt - preWalkCount) + " (재시작 후 첫 동기화: 꺼진 시간분/이전 누적 오차 포함, 알림 없음)");

    sig.barsElapsedSinceRegistration = chartCnt; // 기준값으로 덮어씀(올리기/내리기 모두)

}

// [신규][2026-09-14][판정경로 통합 - 공통 오케스트레이터] 실시간(updatePendingSignals)과
// 워밍업재판정(resolveHistorically 호출부)이 각자 따로 구현하던 "스냅샷 -> 한 개씩
// keep/resolve 판단 -> pendingSignals 재배열 -> 감사호출 -> 파일기록" 뼈대를 하나로
// 합친 공용 함수. 실제 판단로직(가격게이트, 확인단계, 봉replay 등 도메인별 세부사항)은
// 그대로 각 호출부의 resolveFn 콜백 안에 있고, 여기선 그 결과만 받아 공통 뒷정리만
// 담당한다.
// - filterFn(sig): 이 패스의 대상인지(예: 워밍업은 tf일치만, 실시간은 전체)
// - resolveFn(sig, resolvedLines, markExplained): 해소됐으면 resolvedLines에 라인을
//   직접 push하고 recordResolution/logPendingRemoval/markExplained()까지 스스로 호출한
//   뒤 true 반환, 못 정했으면 false 반환(에러처리 방식/카운터증가 등 도메인별 차이는
//   여기서 각자 그대로 유지한다 - 억지로 통일하지 않음)
// - markDirtyOnKeep: 원래 실시간루프는 매번 keepPending에서 excelMarkDirty를 불렀고
//   워밍업루프는 안 불렀던 기존 차이를 그대로 보존하기 위한 플래그.
function runPendingResolutionPass(filterFn, resolveFn, siteName, markDirtyOnKeep) {
    var stillPending = [], resolvedLines = [];
    function keepPending(sig) { stillPending.push(sig); if (markDirtyOnKeep) excelMarkDirty(sig); }
    var beforeSnapshot = pendingSignals.slice();
    var explainedPks = {};
    for (var i = 0; i < pendingSignals.length; i++) {
        var sig = pendingSignals[i];
        if (!filterFn(sig)) { stillPending.push(sig); continue; }
        var resolved;
        try {
            resolved = resolveFn(sig, resolvedLines, function () { explainedPks[pkOf(sig)] = true; });
        } catch (eResolve) {
            // [근본수정][2026-09-25][격리실패 - 신호1개 예외가 배열 전체를 멈추던
            // 버그] 틱봉 v4.11과 동일 원인/동일 수정. 이 for-loop는 신호별
            // try/catch가 없어서, 배열 앞쪽 신호 하나가 resolveFn 처리중
            // 예외를 던지면 그 뒤 모든 신호가 이번 패스에서 통째로 스킵되고
            // 배열 순서가 안 바뀌니 다음 틱/사이클에도 반복돼 영구정지로

            // pk/상태/게이트/에러메시지+스택을 진단로그(RESOLVE_EXCEPTION)와
            // 신호파일에 즉시 남긴다.
            resolved = false;
            try {
                var pkErr = pkOf(sig);
                var errMsg = (eResolve && eResolve.message) ? eResolve.message : String(eResolve);
                var errStack = (eResolve && eResolve.stack) ? String(eResolve.stack).replace(/\n/g, " | ").slice(0, 400) : "";
                logDiagnostic("[DIAG][RESOLVE_EXCEPTION] site=" + siteName + " pk=" + pkErr + " state=" + sig.state + " gate=" + (sig.barsElapsedSinceRegistration || 0) + "/" + REQUIRED_BAR_DELAY + " err=" + errMsg + " stack=" + errStack);

            } catch (eLog) {}
        }
        if (!resolved) keepPending(sig);
    }
    pendingSignals = stillPending;
    auditPendingReassignment(beforeSnapshot, pendingSignals, explainedPks, siteName);
    if (resolvedLines.length > 0) { writeLinesToFile(resolvedLines); pendingStateDirty = true; }
}

function formatHistoryWhen(whenStr) { if (typeof whenStr !== "string") return whenStr; var m = whenStr.match(/^(\d{4})(\d{2})(\d{2}) (.+)$/); return m ? (m[2] + "/" + m[3] + " " + m[4]) : whenStr; }
// [2026-08-17 이식][2안 틱봉 2026-08-13 근본원인수정을 1안에도 동일 적용]
// 예전엔 무조건 배열 끝에 push만 해서, "언제 pushStateHistory가
// 호출됐는지(wall-clock 호출순서)"와 "그 항목이 실제로 가리키는
// 시각(embedded epoch)"이 다르면 순서가 뒤섞였다. 이제 매번 끝에 push하는
// 대신, 각 항목의 실제 시각(epoch)을 기준으로 항상 올바른 시간순 위치에
// 삽입한다 - 같은 epoch끼리는 기존 호출순서를 그대로 유지(안정정렬).
function pushStateHistory(sig, whenStr, stateWord, price, extraDetail) {
    if (!sig.stateHistory) sig.stateHistory = [];
    var dw = formatHistoryWhen(whenStr);
    var epoch = null;
    if (typeof whenStr === "string") {
        var wsParts = whenStr.split(" ");
        if (wsParts.length === 2 && wsParts[0].length === 8) { try { epoch = dateTimeToEpochApprox(wsParts[0], wsParts[1]); } catch (e) { epoch = null; } }
    }
    // [이식][2026-09-26][v5.3] 틱봉엔 있고 분봉엔 없던 "[fn=...,gate=.../26bars]"
    // 판정근거 태그(어느 함수가, 몇 봉째에 이 판정을 내렸는지) - 사용자가 분봉
    // 달성신호 모달에서 이 태그가 빠져 틱봉과 다르게 보이는 걸 지적, 코드
    // 대조로 분봉에만 이 이식이 누락돼있었음을 확인. 틱봉과 동일하게 이식.
    var entry = { t: dw, s: stateWord + "(" + dw + ")@" + Number(price).toFixed(2) + (extraDetail ? "[" + extraDetail + "]" : ""), epoch: epoch };
    // [2026-09-02][근본버그 대칭수정] 2안이 실행테스트로 확인한 버그(호출부가
    // "배열 끝=방금 넣은 것"이라고 위치로 가정하면, 시각기준 정렬삽입 때문에
    // 실제로는 다른 위치에 들어가 엉뚱한 항목을 덮어쓸 수 있음)를 1안도 그대로
    // 갖고 있었다(RETRACTION_BASIS 갱신부, 아래 참고) - 반환값(객체 참조)으로
    // 직접 찾아쓰도록 통일.
    if (epoch === null || sig.stateHistory.length === 0) { sig.stateHistory.push(entry); return entry; }
    var insertAt = sig.stateHistory.length;
    for (var i = sig.stateHistory.length - 1; i >= 0; i--) {
        var existingEpoch = sig.stateHistory[i].epoch;
        if (existingEpoch === null || existingEpoch === undefined || existingEpoch <= epoch) { insertAt = i + 1; break; }
        insertAt = i;
    }
    sig.stateHistory.splice(insertAt, 0, entry);
    return entry;
}
function buildResolvedReasonField(sig, result, whenStr, price, cellTextOverride) {
    var shortReason;
    if (cellTextOverride) shortReason = cellTextOverride;
    else if (result === "ACHIEVED") shortReason = "ACHIEVED(" + whenStr + ")@" + Number(price).toFixed(2);
    else if (result === "INVALID") {
        var rr = sig.resolveReason || "";
        if (rr.indexOf("NO_PULLBACK") === 0) shortReason = "NO_PULLBACK_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else if (rr.indexOf("NO_REBOUND") === 0) shortReason = "NO_REBOUND_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else if (rr.indexOf("AMBIGUOUS_BAR") === 0) shortReason = "AMBIGUOUS_INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
        else shortReason = "INVALID(" + whenStr + ")@" + Number(price).toFixed(2);
    } else shortReason = String(result);
    var histStr = (sig.stateHistory && sig.stateHistory.length > 0) ? " / HISTORY:" + sig.stateHistory.map(function (h) { return h.s; }).join(" -> ") : "";
    return (shortReason + histStr).replace(/,/g, ";");
}
function applyPriceToSignal(sig, curPrice, curDateStr, curTimeStr, gateOpen, callerTag) {
    var whenStr = (curDateStr && curTimeStr) ? (curDateStr + " " + curTimeStr) : "?";
    // [신규][2026-09-29][게이트 신뢰성 안전장치] 위 REQUIRED_BAR_DELAY 옆
    // 주석 참고 - 게이트가 열렸다고 판단됐어도 최근 실측 봉간격 대비
    // 터무니없이 빠르면 "아직 안 열린 것"으로 안전하게 되돌려서 확정을
    // 보류한다(기존 !gateOpen 분기를 그대로 재사용 - 새 상태전이 경로 없음).
    if (gateOpen === true && !isGateCountSane(sig, curDateStr, curTimeStr)) {
        logGateSanitySuspectOnce(sig, whenStr);
        gateOpen = false;
    }
    var gateDetail = "fn=" + (callerTag || "?") + ",gate=" + (sig.barsElapsedSinceRegistration || 0) + "/" + REQUIRED_BAR_DELAY + "bars"; // [이식][2026-09-26][v5.3] 틱봉 동일 기능 이식
    if (gateOpen === false) {
        if (sig.isBull) {
            if (sig.state === "WAITING" && curPrice < sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; if (!sig._pullbackLogged) { sig._pullbackLogged = true; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); } }
            else if (sig.state === "PULLBACK" && curPrice > sig.price) { sig.state = "WAITING"; } // [수정][2026-09-29] 게이트 닫힘 구간의 눌림<->대기 왕복은 이력을 무한히 늘리므로 상태값만 바꾸고 이력은 남기지 않는다(눌림확인은 신호당 첫 1회만 기록)
        } else {
            if (sig.state === "WAITING" && curPrice > sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; if (!sig._pullbackLogged) { sig._pullbackLogged = true; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); } }
            else if (sig.state === "PULLBACK" && curPrice < sig.price) { sig.state = "WAITING"; }
        }
        return null;
    }
    if (sig.isBull) {
        if (sig.state === "WAITING") {
            if (curPrice < sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); }
            else if (curPrice > sig.price) { sig.resolveReason = "NO_PULLBACK->INVALID(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "INVALID_CONFIRMED", curPrice, gateDetail); return "INVALID"; }
        } else if (sig.state === "PULLBACK") {
            if (curPrice >= sig.price) { sig.resolveReason = "PULLBACK->ACHIEVED(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "ACHIEVED_CONFIRMED", curPrice, gateDetail); return "ACHIEVED"; }
        }
    } else {
        if (sig.state === "WAITING") {
            if (curPrice > sig.price) { sig.state = "PULLBACK"; sig.pullbackWhen = whenStr; sig.pullbackPrice = curPrice; pushStateHistory(sig, whenStr, "PULLBACK_CONFIRMED", curPrice); }
            else if (curPrice < sig.price) { sig.resolveReason = "NO_REBOUND->INVALID(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "INVALID_CONFIRMED", curPrice, gateDetail); return "INVALID"; }
        } else if (sig.state === "PULLBACK") {
            if (curPrice <= sig.price) { sig.resolveReason = "REBOUND->ACHIEVED(" + whenStr + "@" + curPrice.toFixed(2) + ")"; pushStateHistory(sig, whenStr, "ACHIEVED_CONFIRMED", curPrice, gateDetail); return "ACHIEVED"; }
        }
    }
    return null;
}

function findFirstBarIndexAtOrAfterEpoch(bars, epoch) {
    var lo = 0, hi = bars.length;
    while (lo < hi) {
        var mid = (lo + hi) >> 1;
        var midEpoch = dateTimeToEpochApprox(String(bars[mid].sdate), bars[mid].stime);
        if (midEpoch < epoch) lo = mid + 1; else hi = mid;
    }
    return lo;
}

    // 분리 자체는 유지한다(WAITING/PULLBACK 코스메틱 이력 중복방지 목적).
    // [근본수정][2026-09-29][WARM 안전망 영구봉쇄 버그] 그런데 그 분리 이후에도
    // 이 판정커서를 "달성/무효(터미널) 판정 스킵" 기준으로 그대로 썼다 - 그런데
    // processLiveBar(실시간 봉완결)는 이 커서를 매 완결봉마다 "지금"까지
    // 무조건 전진시킨다(가격판정 없이, 카운팅커서와 함께). 그 결과 실시간
    // 추적이 살아있는 신호는(항상 그렇다 - REALTIME_TRACK_ENABLED 상시 true)
    // 이 커서가 늘 "방금"까지 따라붙어서, WARM이 실제 차트를 재조회해도
    // startIndex가 배열 끝을 넘어가 이 for문이 통째로 한 번도 안 돈다.
    // 실시간 판정(updatePendingSignals)은 그 순간 들어온 틱 가격 1개만 보는데,
    // 그 틱이 하필 눌림반등/달성 문턱을 놓치면(체결이 뜸한 시간대) - 그 구간의
    // "진짜 high/low"를 재확인할 기회가 영구히 사라진다(NASDAQ100 분봉
    // M0016/M0017 실사례를 정확한 조건으로 재현 완료 - 등록가를 한참 뚫고
    // 내려간 실제 차트 데이터를 갖고 있어도 달성처리가 전혀 안 됨).
    // 판정커서는 "WAITING/PULLBACK 코스메틱 토글 재생만" 막는 용도로 한정하고,
    // ACHIEVED/INVALID 확정 검사는 매 WARM 재조회마다 signalEpoch부터 매번
    // 그 봉의 실제 high/low로 다시 확인한다 - 이미 터미널 처리된 신호는
    // pendingSignals에서 빠져있어 재확인 대상이 아니므로(멱등), 중복확정
    // 위험이 없다. 코스메틱 이력 중복방지(원래 09-25/26 수정의 진짜 목적)는
    // alreadyCosmeticJudged 플래그로 그대로 보존한다(재현테스트로 중복 재발
    // 없음 확인).
    var judgedFromEpoch = (sig.lastJudgedBarEpoch !== undefined && sig.lastJudgedBarEpoch !== null) ? sig.lastJudgedBarEpoch : sigEpochCached;
    var countedFromEpoch = (sig.lastAppliedBarEpoch !== undefined && sig.lastAppliedBarEpoch !== null) ? sig.lastAppliedBarEpoch : sigEpochCached;
    var startBarIndex = findFirstBarIndexAtOrAfterEpoch(bars, sigEpochCached);
    // [신규][v5.17] 재조회 창이 신호 시점을 덮으면, 이 봉 시점까지 센 차트 완결봉 수(trueBarCountAsOfThisBar)를 영속 카운터에도 그대로 기록 -
    // 판정사유(gate=N/26)/게이트 신뢰성 검사/실시간 게이트가 전부 같은 기준값을 본다. 못 덮으면(창이 짧음) 기존 커서 방식 유지.
    var windowCovers = (bars.length > 0 && dateTimeToEpochApprox(String(bars[0].sdate), bars[0].stime) <= sigEpochCached);
    var barsScannedSinceSignal = sig.barsElapsedSinceRegistration || 0;
    // [근본수정][2026-09-29][2차][stale 게이트값을 과거봉에 잘못 적용하던 버그
    // - 재현테스트로 확인] barsScannedSinceSignal은 여러 재조회에 걸쳐
    // 누적되는 "영속" 카운터라, 위에서 항상 signalEpoch부터 다시 스캔하게
    // 바꾼 뒤로는 실시간이 이미 훨씬 뒤(미래) 봉까지 카운트를 올려놓은
    // 상태에서 "과거의" 봉을 재검사할 때도 이 부풀려진 최종값을 그대로
    // 써버려, 실제로는 게이트가 열리기 한참 전인 초기 봉이 "게이트 열림"
    // 판정을 받는 조기확정 위험이 있었다(틱봉 재현: gate=60으로 부풀려진
    // 상태에서 2번째 봉이 즉시 AMBIGUOUS_BAR로 확정되는 것까지 실제 재현
    // 확인). trueBarCountAsOfThisBar는 이번 호출에서 signalEpoch부터 매번
    // 처음부터 다시 세는 "이 봉 시점 기준" 카운트로, 영속 카운터
    // (barsScannedSinceSignal, 대시보드 표시/실시간 연계용, 용도 그대로 유지)와
    // 분리해서 게이트 열림 판정에만 쓴다.
    var trueBarCountAsOfThisBar = 0;
    for (var bj = startBarIndex; bj < bars.length; bj++) {
        var bar = bars[bj];
        var barEpoch = dateTimeToEpochApprox(String(bar.sdate), bar.stime);
        // [2026-09-25][워밍업-실시간 게이트카운트 1봉 불일치 수정, v4.11과 동일]
        // 신호형성봉 자체는 게이트/판정 대상에서 항상 제외(<=로 건너뜀) -
        // 실시간 경로(processLiveBar)와 동일 기준으로 통일된 상태 유지.
        if (barEpoch <= sigEpochCached) continue;
        // [2026-09-25 근본수정][filled(가짜/보간)봉이 게이트+판정에 섞이던 버그]
        // bar.filled=true는 재조회 응답에 구멍(누락)이 생겨 직전 정상봉의
        // 고가/저가를 그대로 복사해 채운 "실제 관측이 아닌" 값이다(Main_OnRcvChartEx
        // 주석 참고) - 다수결/이력 검증(barCheckHistory) 쪽은 이미 filled를
        // 제외하고 있었는데(cbar.filled continue), 정작 여기(26봉 게이트 카운트
        // 및 달성/무효 판정)는 필터링이 없어서 가짜봉도 그대로 카운트되고 그
        // 가짜 고가/저가로 달성/무효가 확정될 수 있었다(사용자 지적 + 재현
        // 테스트로 확인 - 실제관측 6개뿐인데 가짜봉 20개를 더해 게이트가
        // 열리고 마지막 가짜봉 값으로 판정이 나는 것까지 재현됨). 실제 새
        // 데이터가 아니므로 게이트 카운트도, 가격판정도 전부 건너뛴다 -
        // 커서도 안 움직이므로 다음 재조회에서 이 구간이 실제 데이터로
        // 채워지면 그때 다시 정상 처리된다.
        if (bar.filled) continue;
        trueBarCountAsOfThisBar++;
        var alreadyCosmeticJudged = (barEpoch <= judgedFromEpoch); // [2026-09-29] 코스메틱(WAITING/PULLBACK 토글) 이력 중복방지 전용 플래그
        if (barEpoch > countedFromEpoch) {
            barsScannedSinceSignal++;
            sig.barsElapsedSinceRegistration = barsScannedSinceSignal;
            sig.lastAppliedBarEpoch = barEpoch;
            countedFromEpoch = barEpoch;
        }
        if (barEpoch > judgedFromEpoch) { sig.lastJudgedBarEpoch = barEpoch; judgedFromEpoch = barEpoch; }
        if (windowCovers) sig.barsElapsedSinceRegistration = trueBarCountAsOfThisBar;
        var barsElapsedOk = (trueBarCountAsOfThisBar >= REQUIRED_BAR_DELAY); // [2026-09-29] 영속카운터 대신 이 봉 시점 기준 카운트 사용
        var realElapsedMin = (barEpoch - sigEpochCached) / 60;
        var gateOpen = barsElapsedOk && (realElapsedMin >= REQUIRED_BAR_DELAY * sig.period);
        var firstPrice, secondPrice;
        if (sig.isBull) { firstPrice = bar.low; secondPrice = bar.high; } else { firstPrice = bar.high; secondPrice = bar.low; }
        if (!gateOpen) {
            // [2026-09-29] 게이트가 아직 닫혀있는 구간의 WAITING/PULLBACK
            // 코스메틱 토글은, 실시간이 이미 훑은 구간(alreadyCosmeticJudged)이면
            // 재생하지 않는다 - 09-25/26 수정의 원래 목적(중복이력 방지) 그대로 보존.
            if (!alreadyCosmeticJudged) { applyPriceToSignal(sig, firstPrice, bar.sdate, bar.stime, false, "resolveHistorically(WARM)"); applyPriceToSignal(sig, secondPrice, bar.sdate, bar.stime, false, "resolveHistorically(WARM)"); }
            continue;
        }
        if (sig.state === "WAITING" && bar.low < sig.price && bar.high > sig.price) {
            sig.resolveReason = "AMBIGUOUS_BAR(" + bar.sdate + "_" + bar.stime + ")->INVALID";
            pushStateHistory(sig, bar.sdate + " " + bar.stime, "INVALID_CONFIRMED", sig.price);
            return { result: "INVALID", sdate: bar.sdate, stime: bar.stime, reason: sig.resolveReason };
        }
        var stateBeforeThisBar = sig.state;
        // [2026-09-14][죽은코드 제거] 확인단계(confirmHistoricalResult)가 항상
        // 즉시통과라, 스냅샷/롤백 없이 결과가 나오면 그대로 반환한다.
        var r1 = applyPriceToSignal(sig, firstPrice, bar.sdate, bar.stime, true, "resolveHistorically(WARM)");
        if (r1) return { result: r1, sdate: bar.sdate, stime: bar.stime, reason: sig.resolveReason };
        if (stateBeforeThisBar === "WAITING" && sig.state === "PULLBACK") continue;
        var r2 = applyPriceToSignal(sig, secondPrice, bar.sdate, bar.stime, true, "resolveHistorically(WARM)");
        if (r2) return { result: r2, sdate: bar.sdate, stime: bar.stime, reason: sig.resolveReason };
    }
    return null;
}

function isDuplicatePendingSignal(itemName, tf, kind, sdate, stime, price, trueSdate, trueStime, leftFlatRun, apexHeight) {
    // [근본수정][2026-09-13][최종 확정 - 시간/위치가 아니라 모양값으로만
    // 판정] 시간허용오차(과거)와 캐시위치인접성(오늘 시도) 둘 다 "봉 하나
    // 건너 우연히 같은 세기/가격/타임프레임으로 생기는 진짜 새 신호"까지
    // 잘못 걸러낼 수 있다는 지적을 받고 최종 기각됨 - 봉의 시간이나
    // 위치에 기대는 방식은 전부 같은 함정(그 봉이 재구성 때마다 흔들림)
    // 을 벗어나지 못한다. apexHeight(꼭지점 높이)/leftFlatRun(선행평탄
    // 봉수)는 봉의 위치가 아니라 실제 가격관계에서 나온 값이라, 같은
    // 진짜 삼각형이면 어느 봉이 확인/꼭지점으로 뽑히든 값 자체는 거의
    // 그대로 나온다 - 이 값으로만 판정한다: tf/kind/price 동일 +
    // apexHeight 완전동일 + leftFlatRun 차이 1 이내, 전부 만족해야 중복.
    // 값 정보가 없는(구버전 호출) 경우에만 trueSdate 완전일치로 대체.
    if (apexHeight !== undefined && apexHeight !== null && leftFlatRun !== undefined && leftFlatRun !== null) {
        for (var pj = 0; pj < pendingSignals.length; pj++) {
            var pp = pendingSignals[pj];
            if (pp.itemName !== itemName || pp.tf !== tf || pp.kind !== kind || Math.abs(price - pp.price) >= 0.01) continue;
            if (pp.apexHeight === undefined || pp.apexHeight === null || pp.leftFlatRun === undefined || pp.leftFlatRun === null) continue;
            if (pp.apexHeight === apexHeight && Math.abs(pp.leftFlatRun - leftFlatRun) <= 1) return true;
        }
        return false;
    }
    if (trueSdate && trueStime) {
        for (var pj2 = 0; pj2 < pendingSignals.length; pj2++) {
            var pp2 = pendingSignals[pj2];
            if (pp2.itemName === itemName && pp2.tf === tf && pp2.kind === kind &&
                pp2.trueSdate === trueSdate && pp2.trueStime === trueStime) return true;
        }
        return false;
    }
    var thisEpoch = dateTimeToEpochApprox(String(sdate), stime);
    for (var pi = 0; pi < pendingSignals.length; pi++) {
        var p = pendingSignals[pi];
        if (p.itemName === itemName && p.tf === tf && p.kind === kind && Math.abs(p.price - price) < 0.01) {
            var pEpoch = (p.signalEpoch !== undefined && p.signalEpoch !== null) ? p.signalEpoch : dateTimeToEpochApprox(String(p.sdate), p.stime);
            if (Math.abs(thisEpoch - pEpoch) <= DUP_SIGNAL_TIME_TOLERANCE_SEC) return true;
        }
    }
    return false;
}
function findPendingMatch(itemName, tf, kind, sdate, stime, price) {
    var thisEpoch = dateTimeToEpochApprox(String(sdate), stime);
    for (var pi = 0; pi < pendingSignals.length; pi++) {
        var p = pendingSignals[pi];
        if (p.itemName === itemName && p.tf === tf && p.kind === kind && Math.abs(p.price - price) < 0.01) {
            var pEpoch = (p.signalEpoch !== undefined && p.signalEpoch !== null) ? p.signalEpoch : dateTimeToEpochApprox(String(p.sdate), p.stime);
            if (Math.abs(thisEpoch - pEpoch) <= DUP_SIGNAL_TIME_TOLERANCE_SEC) return p;
        }
    }
    return null;
}
// [2026-08-13 리팩터] 실시간(processLiveBar)과 워밍업/재조회
// (Main_OnRcvChartEx) 두 등록경로 모두 이 함수 하나만 거치도록 통일했다.
// [근본수정: 단일 진실원] 신호 객체 생성 + REGISTERED 이력 기록을 이 함수
// 하나로 통일. trial(즉시판정) 경로와 registerPendingSignal(대기등록)
// 경로가 반드시 이 함수만 거치도록 해서 케이스별 누락/정책불일치가
// 구조적으로 재발하지 않게 한다. registeredWhen: trial 경로는 신호 자체
// 봉시각을, registerPendingSignal 경로는 nowServerTimeStr()(현재
// 처리시각)를 넘긴다.
function buildNewSignalObject(itemName, tf, kind, sdate, stime, price, period, registeredWhen, source, leftFlatStar, trueSdate, trueStime, leftFlatRun, apexHeight) {
    var regWhenForHistory = registeredWhen || (sdate + " " + stime);
    // [2026-09-02][등록경로 태그 보완] 2안과 동일 패턴 - 웜업(WARM)/실시간(REAL)
    // 등록경로를 stateHistory[0]에 즉시 남겨, 크래시 등으로 나중에 유실돼도
    // 등록 순간의 경로를 텍스트에서 바로 확인 가능하게 한다(추정 불필요).
    var sourceTag = source ? "[" + source + "]" : "";
    return {
        itemName: itemName, tf: tf, kind: kind, sdate: sdate, stime: stime, price: price,
        isBull: kind.indexOf("BULL") === 0, state: "WAITING", period: period,
        barsElapsedSinceRegistration: 0,
        // [신규][2026-09-07][틱봉과 동일 이식] 판정/철회에도 신호세기 기반
        // 확인단계를 적용하려면 이 값이 신호 객체 자체에 남아있어야 한다.
        leftFlatStar: leftFlatStar || 0,
        // [근본수정][2026-09-13][중복판정 신원기준 최종 - 봉의 시간/위치가
        // 아니라 값(모양) 기준] 확인봉(trueSdate)도 꼭지점봉(apexSdate)도
        // 결국 "어느 봉인지"(위치/시각)를 특정하는 방식이라 재구성 흔들림
        // 자체를 벗어나지 못한다는 지적(사용자) - leftFlatRun(선행평탄
        // 봉수)/apexHeight(꼭지점 높이)는 봉 위치가 아니라 실제 가격관계
        // 에서 나온 값이라, 같은 진짜 삼각형이면 어느 봉이 확인/꼭지점으로
        // 뽑히든 값 자체는 거의 그대로 나온다. 이 값들을 저장해서
        // isDuplicatePendingSignal이 시간/위치가 아니라 모양값으로만
        // 판정하게 한다.
        leftFlatRun: leftFlatRun, apexHeight: apexHeight,
        lastAppliedBarEpoch: null,
        // [신규][2026-09-25][카운팅커서/판정커서 분리] resolveHistorically
        // 전용 판정커서 - null이면 signalEpoch부터 다시 판정을 시도한다
        // (기존 신호 복원시 이 필드가 없으면 자동으로 이 폴백이 걸려서,
        // 과거에 lastAppliedBarEpoch만으로 인해 재판정을 놓쳤던 신호도
        // 다음 재워밍업에서 처음부터 다시 판정된다).
        lastJudgedBarEpoch: null,
        // [긴급수정][2026-09-07][백데이팅과 게이트카운트 혼용 버그](틱봉과
        // 동일 원인/동일 수정) 되짚기(sdate/stime, 화면표시/중복방지 전용)를
        // 게이트(signalEpoch) 계산에도 그대로 써버린 게 버그였다 - 반드시
        // trueSdate/trueStime(백데이팅 안 된 진짜 삼각형완성 시각)으로 계산.
        signalEpoch: dateTimeToEpochApprox(String(trueSdate || sdate), trueStime || stime), registeredAt: Date.now(),
        // [근본수정][2026-09-09][trueSdate/trueStime을 신호객체 필드로도
        // 저장] signalEpoch 계산에만 쓰고 정작 객체 자체엔 안 남기고
        // 있었다 - 그래서 isDuplicatePendingSignal이 이 값을 쓰도록
        // 고쳐도(sg.trueSdate 참조) 항상 undefined라 실질적으로 무효였음.
        // 진짜 원본시각 기준 중복판정이 작동하려면 이 필드가 반드시 있어야 함.
        trueSdate: trueSdate || sdate, trueStime: trueStime || stime,
        lastCheckedAt: null, lastCheckedCycle: null,
        lastCheckedReason: "REGISTERED_NOT_YET_RECHECKED(1안: OnBarAppended로만 추적)",
        stateHistory: [{ t: formatHistoryWhen(regWhenForHistory), s: "REGISTERED(" + formatHistoryWhen(regWhenForHistory) + ")" + sourceTag + "@" + Number(price).toFixed(2) }],
        _saveConfirmed: false
    };
}

    // 자연스럽게 쓰이게 한다 - 이러면 눌림확인(항상 신호형성시각 이후
    // 봉만 재생)이 등록보다 항상 뒤에 오는 게 구조적으로 보장된다.

    // 명시적으로 적용한다.
    var regWhenFallback = sdate + " " + stime;

    // 있었음), 정작 이 줄은 그 필드를 한 번도 실어보낸 적이 없었다 -
    // 8번째 필드가 없으면 대시보드가 sig.kind(="SAVE_SCHEDULED"라는

                        // 등록 성공 여부와 무관하게 항상 파일엔 기록됐다. 등록시점
                        // 안정성게이트(WARM 전용)를 넣은 뒤로, 게이트가 "아직
                        // 불안정하니 보류"하며 false를 반환해도 이미 파일엔 줄이
                        // 쓰여있어서, 대시보드엔 "방금 등록된 신호"가 보이는데 정작
                        // pendingSignals(감시목록)엔 없어서 바로 다음 하트비트부터
                        // "??감시목록 확인 안됨"으로 뜨는 유령신호가 실제로 발생했다
                        // (사용자 실제사례: M0003). 등록을 먼저 시도해서 성공했을
                        // 때만 파일에 줄을 쓰도록 순서를 뒤집는다 - 게이트가 보류하면
                        // 이번 사이클엔 파일에도 아무 흔적을 안 남기고, 다음 사이클에
                        // (안정되면) 자연스럽게 다시 시도된다.
                        var registeredNow = registerPendingSignal(thisItemName, thisFileLabel, signals[s].kind, writeSdate, writeStime, signals[s].price, thisCycle, "WARM" + frameShiftNote, signals[s].leftFlatStar, signals[s].trueSdate, signals[s].trueStime, signals[s].leftFlatRun, signals[s].apexHeight);
                        if (registeredNow) newSignalLines.push(ln);
                    }
                }
            } catch (e) {}
        }
        writeLinesToFile(newSignalLines);

        // [제거][사용자 요청][철회기능 전체 제거] 여기 있던 철회판정 루프(마진기반 철회/요청봉수부족 감시목록제외 포함)를 통째로 삭제함 - pendingSignals는 이제 이 사이클에서 줄어들지 않음.

        if (bars.length > 0) {
            var lastWarmupBar = bars[bars.length - 1];
            logDiagnostic("[DIAG][CYCLE_DONE] label=" + thisFileLabel + " 마지막봉 high=" + lastWarmupBar.high + " low=" + lastWarmupBar.low, lastWarmupBar.sdate, lastWarmupBar.stime);
            var newestEpochThisCycle = dateTimeToEpochApprox(String(lastWarmupBar.sdate), lastWarmupBar.stime);
            var prevInfo = lastCycleBarInfo[thisFileLabel];
            if (prevInfo && dateTimeToEpochApprox(String(bars[0].sdate), bars[0].stime) > prevInfo.newestEpoch) {
                logDiagnostic("[DIAG][ADAPTIVE_GAP_DETECTED] label=" + thisFileLabel + " 공백감지 - 다음 요청이 자동 확대됩니다");
            }
            lastCycleBarInfo[thisFileLabel] = { newestEpoch: newestEpochThisCycle };
        }
        // [2026-09-25 근본수정][침묵 완료처리 버그] 예전엔 completedCycles[thisCycle]=true가
        // 이 try/catch 블록 "바깥"에 있어서, 처리 도중 예외가 나도(catch로 빠져도)
        // 무조건 완료 처리되고 있었다 - 재시도/저품질(degraded) 어느 경로도 안 거치고
        // 그냥 조용히 "다음 순환까지 방치"되는 사각지대였다(틱봉 파일은 처음부터
        // 이 라인이 try 안, 성공 경로에만 있어서 이 문제가 없었음 - 틱봉과 동일하게 맞춤).
        // 이제 성공했을 때만(예외 없이 여기까지 도달) 완료 처리한다.
        _rcvOk = (bars.length > 0); // [2026-09-25 신규] 위에서 이미 예외 없이 왔다는 뜻 - 추가로 봉을 1개 이상 실제로 받았을 때만 true
        // [근본수정][2026-09-29][침묵 완료처리 사각지대] 예전엔 예외만 없으면
        // 봉을 0개 받았어도(_rcvOk===false) completedCycles=true로 찍어서
        // "완료"로 넘어갔다 - 재시도큐에도 안 들어가고 저품질(degraded)로도
        // 안 넘어가는 사각지대라, [진행상황]/[순환완료] 로그의 분자 합이
        // 총원(110)과 안 맞는 걸로 실사례(M0001~M0007, 104/110)로 확인됨.
        // 봉이 0개면 예외와 똑같이 "실패"로 보고 재시도큐로 보낸다 -
        // recordFailureForRetry가 기존 재시도횟수 카운팅을 그대로 하고,
        // MAX_RETRY_ATTEMPTS를 다 채우면 그 함수 스스로 degradedPeriods+
        // completedCycles를 true로 만든다(기존 저품질 전환 로직 재사용,
        // 새 분기 없음).
        if (_rcvOk) {
            completedCycles[thisCycle] = true;
        } else {
            recordFailureForRetry({ code: MARKET_DATA_CODE, nameEn: thisItemName, cycle: thisCycle });
        }
    } catch (e) {
        logMsg("  !! 처리오류: " + thisFileLabel + " - " + e.message);
        // [2026-09-25 근본수정] 완료처리 대신 재시도큐로 - 틱봉 파일의 기존 catch와 동일 패턴
        recordFailureForRetry({ code: MARKET_DATA_CODE, nameEn: thisItemName, cycle: thisCycle });
    }

    pendingRemoveObjects.push(chartEx);
    // [2026-08-13 안전성수정] 삭제타이머는 이제 Main_OnTimer에서 스스로 재무장하므로 여기서 별도 SetTimer 불필요
    if (_rcvOk) filledCycles[thisCycle] = true; // [2026-09-25 신규] try가 예외 없이 끝까지 실행되고 봉을 1개 이상 받았을 때만 true - thisCycle은 #주기 프레임재조준이 있었으면 이미 실제값으로 치환된 상태라 그대로 정확함

    pacingOnCycleSuccess();
    currentIndex++;
    if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
    // [이식][2026-09-10][틱봉에만 있던 워밍업 진행상황 로그를 분봉에도
    // 추가] 틱봉엔 2026-09-05부터 있던 기능 - 워밍업 1바퀴가 다 끝나야
    // 하트비트가 시작되는데, 그 전까지는 개별 요청 로그가 화면에 안
    // 뜨는 조용한 태그라 실제로는 정상 진행 중이어도 멈춘 것처럼 보일
    // 수 있다. 10개 완료마다 눈에 보이는 진행상황을 남긴다.
    // [2026-09-25 수정][실제채움 기준으로 정정] completedCycles(성공+degraded
    // 포기 합산)를 그대로 "실제 채워진 개수"라고 표시하던 게 거짓정보였음
    // (검증 완료 - 아래 verify_progress_log 계열 테스트 참고). filledCycles
    // (진짜 데이터 받은 것만)를 분자로 쓰고, degraded 포기분은 별도로 명시한다.
    if (currentIndex % 10 === 0 || currentIndex === targetList.length) {
        var _degradedNowCount = Object.keys(degradedPeriods).length;
        logMsg("[진행상황] " + Object.keys(filledCycles).length + "/" + baseTargetList.length + " 타임프레임 실제 데이터 수신 완료" +
            (_degradedNowCount > 0 ? " (그 외 저품질(degraded)로 포기 " + _degradedNowCount + "개 - 완료 처리는 됐으나 실데이터 아님)" : ""));
    }
    Main.SetTimer(1, pacingMsCurrent);
}

function Main_OnUpdateMarket(itemcode, updateID, exchangeKind) {
    if (!marketDataObj || itemcode !== MARKET_DATA_CODE) return;
    if (updateID !== 20001) return;
    totalTicksReceived++;
    var curPrice = Number(marketDataObj.current);
    if (!isFinite(curPrice) || curPrice <= 0) return;
    var sdate = parseDateField(marketDataObj.date), stime = parseTimeField(marketDataObj.time);
    if (!sdate || !stime) return;
    var epochSeconds = dateTimeToEpochApprox(sdate, stime);
    var nowWallMs = Date.now();
    if (lastTickEpoch !== null) {
        var gapSec = epochSeconds - lastTickEpoch;
        if (gapSec > maxObservedTickGapSec) maxObservedTickGapSec = gapSec;
        var wallGapMs = (lastTickWallClockMs !== null) ? (nowWallMs - lastTickWallClockMs) : null;
        if (gapSec > 60 || (wallGapMs !== null && wallGapMs > 60000)) {
            tickGapWarnCount++;
            logDiagnostic("[DIAG][TICK_GAP] 서버시간갭=" + gapSec.toFixed(0) + "s / 내PC벽시계갭=" + (wallGapMs !== null ? (wallGapMs / 1000).toFixed(1) : "?") + "s (누적" + tickGapWarnCount + ")" +
                (wallGapMs !== null && Math.abs(wallGapMs / 1000 - gapSec) > 10 ? " ?불일치큼" : ""), sdate, stime);
        }
    }
    lastTickEpoch = epochSeconds;
    lastTickWallClockMs = nowWallMs;
    if (REALTIME_TRACK_ENABLED) {
        Object.keys(liveBuilders).forEach(function (fileLabel) {
            try {
                var period = parseInt(fileLabel.slice(1), 10);
                var bar = liveBuilders[fileLabel].onTick(curPrice, sdate, stime, epochSeconds);
                if (bar) processLiveBar(fileLabel, period, bar.high, bar.low, bar.sdate, bar.stime, bar.gapBars);
            } catch (e) {}
        });
        try { updatePendingSignals(curPrice, sdate, stime); } catch (e) {}
    }
}

function Main_OnTimer(nEventID) {
    if (nEventID == 1) { Main.KillTimer(1); processNext(); }
    else if (nEventID == 3) {
        Main.KillTimer(3); quickSaveScheduled = false;
        if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
    }
    else if (nEventID == 2) {
        Main.KillTimer(2);
        if (waitingForResponse) {
            waitingForResponse = false;
            var timedOutReq = null;
            for (var qi2 = requestQueue.length - 1; qi2 >= 0; qi2--) if (requestQueue[qi2].seq === activeRequestSeq) { timedOutReq = requestQueue.splice(qi2, 1)[0]; break; }
            // [근본수정][2026-09-30][타임아웃 요청 증발] 타임아웃난 요청을 큐에서 지우기만 하고 재시도큐/저품질 어디에도 안 넣어 [순환완료] 합계에서 증발하던 것 수정.
            if (timedOutReq) {
                logDiagnostic("[DIAG][REQUEST_TIMEOUT] label=" + timedOutReq.fileLabel + " cycle=" + timedOutReq.cycle + " - 재시도큐로 보냄");
                recordFailureForRetry({ code: MARKET_DATA_CODE, nameEn: timedOutReq.itemName, cycle: timedOutReq.cycle });
            }
            timeoutTrackedLabels[currentFileLabel] = true; // [신규][2026-09-10] 이 라벨의 다음 정상매칭까지 추적
            pacingOnCycleFailure();
            currentIndex++;
            Main.SetTimer(1, pacingMsCurrent + TIMEOUT_COOLDOWN_MS);
        }
    } else if (nEventID == 4) {
        Main.KillTimer(4);
        var elapsedMin = scriptStartTime ? Math.floor((Date.now() - scriptStartTime) / 60000) : -1;
        logMsg("[하트비트] 경과:" + elapsedMin + "분 / 대기중:" + pendingSignals.length + " / 달성:" + achievedCount + " / 무효:" + invalidCount +
            " / 완결봉:" + totalBarsCompleted + " / 완료순환:" + totalPassesCompleted + "바퀴");
        logDiagnostic("[DIAG][RESOURCE] pendingSignals=" + pendingSignals.length + " requestQueue=" + requestQueue.length +
            " fileWriteBuffer=" + fileWriteBuffer.length + " diagnosticBuffer=" + diagnosticBuffer.length +
            " seenKeys=" + Object.keys(seenKeys).length + " seenKeysByPrice=" + Object.keys(seenKeysByPrice).length +
            " recentResolutions=" + Object.keys(recentResolutions).length + " orphanedSignalDetails=" + Object.keys(orphanedSignalDetails).length);
        pruneDedupStructures();
        flushDiagnosticLog();
        try {
            var hbDate = parseDateField(marketDataObj ? marketDataObj.date : 0);
            var hbTime = parseTimeField(marketDataObj ? marketDataObj.time : null) || "00:00:00";
            timedPrintOnFile(OUTPUT_FILE, (currentItemName || INSTRUMENT_NAME) + ",HEARTBEAT_MIN,HEARTBEAT," + hbDate + "," + hbTime + ",0," + buildPendingKeysString() + "," + buildPendingDetailsString() + "\n");
        } catch (e) {}
        if (pendingStateDirty) { savePendingState(); pendingStateDirty = false; }
        Main.SetTimer(4, heartbeatIntervalWithJitter());
    } else if (nEventID == 5) {
        // [2026-08-13 안전성수정] 요청 페이싱이 0ms까지 내려간 정상운영
        // 상태(영구순환)에서는 외부 재무장 방식이 삭제타이머를 계속
        // 리셋시켜 Main.RemoveObject가 영원히 안 불릴 수 있었다(틱파일과
        // 동일 위험 - 상세 이유는 틱파일 주석 참고). 이제 이 핸들러가
        // 요청 사이클과 무관하게 스스로 고정주기로 재무장한다.
        if (pendingRemoveObjects.length > 0) {
            var objectsToRemove = pendingRemoveObjects;
            pendingRemoveObjects = [];
            for (var ri5 = 0; ri5 < objectsToRemove.length; ri5++) {
                try { Main.RemoveObject(objectsToRemove[ri5]); }
                catch (e) { logDiagnostic("[DIAG][REMOVE_OBJECT_FAILED] " + e.message); }
            }
            lastRemoveObjectAt = Date.now();
        }
        Main.SetTimer(5, REMOVE_OBJECT_DELAY_MS);
    } else if (nEventID == 7) {
        // [신규][2026-09-09][삭제~다음요청 최소대기용 재시도 타이머]
        Main.KillTimer(7);
        processNext();
    } else if (nEventID == 6) {
        // [2026-09-02][degraded복구 보완] 본 순환을 안 끊고, 진행중 요청이
        // 없을 때만 degraded 라벨 하나를 targetList의 바로 다음 자리에
        // 끼워넣는다. 별도 요청상태/응답매칭 분기가 필요없어(기존
        // processNext/Main_OnRcvChartEx 그대로 재사용) 뒤섞일 여지가 없다.
        if (!waitingForResponse) {
            var degradedList6 = Object.keys(degradedPeriods);
            for (var dgi = 0; dgi < degradedList6.length; dgi++) {
                var dLabel = degradedList6[dgi];
                if (degradedRecoveryPending[dLabel]) continue;
                var dCycle = parseInt(dLabel.slice(1), 10);
                degradedRecoveryPending[dLabel] = true;
                var insertAt = Math.min(currentIndex, targetList.length); // [경계처리] 순환 끝나는 찰나여도 범위밖 삽입 방지
                targetList.splice(insertAt, 0, { code: MARKET_DATA_CODE, nameEn: INSTRUMENT_NAME, cycle: dCycle });
                delete completedCycles[dCycle]; // [신규][사용자 지적으로 발견한 충돌 수정] 포기시 completedCycles에 "임시완료"로 표시해뒀던 걸 진짜 재시도 직전에 해제 - 안 그러면 스킵로직이 이 재삽입을 건너뜀
                logMsg("  ? [degraded복구] " + dLabel + " 본 순환에 재삽입(1분 주기)");
                break; // 한 틱에 하나만(과도한 동시삽입 방지)
            }
        }
        Main.SetTimer(6, DEGRADED_RECOVERY_INTERVAL_MS);
    }
